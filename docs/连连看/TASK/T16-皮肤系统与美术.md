/** 纹理绘制上下文。 */
export interface TextureCtx {
  ctx: CanvasRenderingContext2D;
  x: number; y: number; w: number; h: number;
  /** 确定性随机源（0..1） */
  rand: () => number;
}

/**
 * 2D value noise + fbm。用于大理石与星尘纹理。
 */
export class ValueNoise {
  private perm: Uint8Array;

  constructor(seed: number) {
    const p = new Uint8Array(512);
    const base = new Uint8Array(256);
    for (let i = 0; i < 256; i++) base[i] = i;
    // 确定性洗牌
    let s = seed >>> 0;
    for (let i = 255; i > 0; i--) {
      s = (s * 1664525 + 1013904223) >>> 0;
      const j = s % (i + 1);
      [base[i], base[j]] = [base[j], base[i]];
    }
    for (let i = 0; i < 512; i++) p[i] = base[i & 255];
    this.perm = p;
  }

  /** 2D 梯度噪声，返回 0..1。 */
  noise2(x: number, y: number): number {
    const xi = Math.floor(x) & 255;
    const yi = Math.floor(y) & 255;
    const xf = x - Math.floor(x);
    const yf = y - Math.floor(y);
    const u = xf * xf * (3 - 2 * xf);
    const v = yf * yf * (3 - 2 * yf);
    const hash = (a: number, b: number) => this.perm[(this.perm[a] + b) & 511] / 255;
    const n00 = hash(xi, yi);
    const n10 = hash(xi + 1, yi);
    const n01 = hash(xi, yi + 1);
    const n11 = hash(xi + 1, yi + 1);
    const nx0 = n00 + (n10 - n00) * u;
    const nx1 = n01 + (n11 - n01) * u;
    return nx0 + (nx1 - nx0) * v;
  }

  /** 分形叠加噪声。 */
  fbm(x: number, y: number, octaves = 4): number {
    let sum = 0, amp = 0.5, freq = 1;
    for (let i = 0; i < octaves; i++) {
      sum += this.noise2(x * freq, y * freq) * amp;
      freq *= 2;
      amp *= 0.5;
    }
    return sum;
  }
}

/** 纯色 + 顶部高光。 */
export function texNone(t: TextureCtx, base: string, light: string): void {
  const { ctx, x, y, w, h } = t;
  ctx.fillStyle = base;
  ctx.fillRect(x, y, w, h);
  const g = ctx.createLinearGradient(x, y, x, y + h);
  g.addColorStop(0, light);
  g.addColorStop(0.45, 'rgba(255,255,255,0)');
  ctx.fillStyle = g;
  ctx.fillRect(x, y, w, h);
}

/** 木纹：正弦扰动的深色横纹。 */
export function texWood(t: TextureCtx, base: string, dark: string): void {
  const { ctx, x, y, w, h } = t;
  ctx.fillStyle = base;
  ctx.fillRect(x, y, w, h);
  const lines = 4;
  ctx.strokeStyle = dark;
  ctx.lineWidth = Math.max(1, h * 0.06);
  for (let i = 0; i < lines; i++) {
    const phase = t.rand() * Math.PI * 2;
    const yy = y + h * ((i + 0.5) / lines);
    ctx.beginPath();
    for (let px = 0; px <= w; px += 4) {
      const py = yy + Math.sin(px * 0.12 + phase) * h * 0.07;
      if (px === 0) ctx.moveTo(x + px, py); else ctx.lineTo(x + px, py);
    }
    ctx.stroke();
  }
}

/** 大理石：噪声脉络。 */
export function texMarble(t: TextureCtx, base: string, vein: string): void {
  const { ctx, x, y, w, h } = t;
  ctx.fillStyle = base;
  ctx.fillRect(x, y, w, h);
  const noise = new ValueNoise(Math.floor(t.rand() * 100000));
  ctx.strokeStyle = vein;
  ctx.lineWidth = Math.max(1, h * 0.05);
  for (let i = 0; i < 3; i++) {
    ctx.globalAlpha = 0.35 + t.rand() * 0.3;
    ctx.beginPath();
    const phase = t.rand() * 100;
    for (let px = 0; px <= w; px += 3) {
      const n = noise.fbm(px * 0.02 + phase, i * 3.7, 3);
      const py = y + h * (0.25 + i * 0.25) + (n - 0.5) * h * 0.5;
      if (px === 0) ctx.moveTo(x + px, py); else ctx.lineTo(x + px, py);
    }
    ctx.stroke();
  }
  ctx.globalAlpha = 1;
}

/** 金属拉丝：密集水平细线。 */
export function texBrushed(t: TextureCtx, base: string, dark: string): void {
  const { ctx, x, y, w, h } = t;
  ctx.fillStyle = base;
  ctx.fillRect(x, y, w, h);
  ctx.strokeStyle = dark;
  ctx.lineWidth = 1;
  for (let py = 0; py < h; py += 2) {
    ctx.globalAlpha = 0.1 + t.rand() * 0.25;
    ctx.beginPath();
    ctx.moveTo(x, y + py);
    ctx.lineTo(x + w, y + py);
    ctx.stroke();
  }
  ctx.globalAlpha = 1;
}

/** 冰晶：斜向亮线 + 中心菱形。 */
export function texIce(t: TextureCtx, base: string, light: string): void {
  const { ctx, x, y, w, h } = t;
  ctx.fillStyle = base;
  ctx.fillRect(x, y, w, h);
  ctx.strokeStyle = light;
  ctx.lineWidth = Math.max(1, h * 0.07);
  ctx.globalAlpha = 0.6;
  for (let i = 0; i < 3; i++) {
    const off = t.rand();
    ctx.beginPath();
    ctx.moveTo(x - h * 0.2 + off * w, y + h);
    ctx.lineTo(x + w * (0.4 + off * 0.6), y - h * 0.2);
    ctx.stroke();
  }
  ctx.globalAlpha = 0.5;
  ctx.fillStyle = light;
  ctx.beginPath();
  ctx.moveTo(x + w / 2, y + h * 0.22);
  ctx.lineTo(x + w * 0.72, y + h / 2);
  ctx.lineTo(x + w / 2, y + h * 0.78);
  ctx.lineTo(x + w * 0.28, y + h / 2);
  ctx.closePath();
  ctx.fill();
  ctx.globalAlpha = 1;
}

/** 霓虹：深底 + 发光边框 + 中心亮线。 */
export function texNeon(t: TextureCtx, base: string, glow: string): void {
  const { ctx, x, y, w, h } = t;
  ctx.fillStyle = base;
  ctx.fillRect(x, y, w, h);
  ctx.save();
  ctx.shadowColor = glow;
  ctx.shadowBlur = Math.max(6, h * 0.35);
  ctx.strokeStyle = glow;
  ctx.lineWidth = Math.max(2, h * 0.09);
  ctx.beginPath();
  ctx.moveTo(x + w * 0.18, y + h / 2);
  ctx.lineTo(x + w * 0.82, y + h / 2);
  ctx.stroke();
  ctx.restore();
}

/** 黄金：斜向渐变 + 高光带。 */
export function texGold(t: TextureCtx, base: string, light: string, dark: string): void {
  const { ctx, x, y, w, h } = t;
  const g = ctx.createLinearGradient(x, y, x + w, y + h);
  g.addColorStop(0, dark);
  g.addColorStop(0.4, light);
  g.addColorStop(0.55, base);
  g.addColorStop(0.75, light);
  g.addColorStop(1, dark);
  ctx.fillStyle = g;
  ctx.fillRect(x, y, w, h);
}

/** 星尘：深底 + 星点（带闪烁时间）。 */
export function texStardust(t: TextureCtx, base: string, star: string, timeMs: number): void {
  const { ctx, x, y, w, h } = t;
  ctx.fillStyle = base;
  ctx.fillRect(x, y, w, h);
  const count = 18;
  for (let i = 0; i < count; i++) {
    const px = x + t.rand() * w;
    const py = y + t.rand() * h;
    const twinkle = 0.5 + 0.5 * Math.sin(timeMs / 400 + i * 1.7);
    ctx.globalAlpha = twinkle;
    ctx.fillStyle = star;
    const s = 1 + t.rand() * 1.6;
    ctx.fillRect(px, py, s, s);
  }
  ctx.globalAlpha = 1;
}

/** 星河：多层径向渐变 + 星点。 */
export function texGalaxy(t: TextureCtx, base: string, c1: string, c2: string, timeMs: number): void {
  const { ctx, x, y, w, h } = t;
  ctx.fillStyle = base;
  ctx.fillRect(x, y, w, h);
  const cx = x + w / 2, cy = y + h / 2;
  const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, Math.max(w, h) * 0.7);
  g.addColorStop(0, c1);
  g.addColorStop(0.5, c2);
  g.addColorStop(1, 'rgba(0,0,0,0)');
  ctx.fillStyle = g;
  ctx.fillRect(x, y, w, h);
  const count = 22;
  for (let i = 0; i < count; i++) {
    const px = x + t.rand() * w;
    const py = y + t.rand() * h;
    const twinkle = 0.4 + 0.6 * Math.sin(timeMs / 500 + i * 2.3);
    ctx.globalAlpha = twinkle;
    ctx.fillStyle = '#FFFFFF';
    ctx.fillRect(px, py, 1.2, 1.2);
  }
  ctx.globalAlpha = 1;
}

  return SKINS.find((s) => s.id === id) ?? SKINS[0];

/** 由 HSL 生成一组派生色。 */
export interface SkinPalette {
  base: string;
  light: string;
  dark: string;
  border: string;
  glow: string;
}

/** 根据皮肤生成调色板。 */
export function paletteOf(skin: Skin): SkinPalette {
  const { hue, sat, light } = skin;
  return {
    base: `hsl(${hue}, ${sat}%, ${light}%)`,
    light: `hsl(${hue}, ${sat}%, ${Math.min(95, light + 22)}%)`,
    dark: `hsl(${hue}, ${sat}%, ${Math.max(8, light - 22)}%)`,
    border: `hsl(${hue}, ${Math.min(100, sat + 10)}%, ${Math.max(10, light - 32)}%)`,
    glow: `hsl(${hue}, ${sat + 15}%, ${Math.min(85, light + 30)}%)`,
  };
}

/** 确定性伪随机（同一格每次渲染一致）。 */
function makeRand(seed: number): () => number {
  let s = seed >>> 0;
  return () => {
    s = (s * 1664525 + 1013904223) >>> 0;
    return s / 4294967296;
  };
}

/**
 * 在指定矩形内绘制皮肤纹理。
 */
export function paintSkin(
  ctx: CanvasRenderingContext2D,
  skin: Skin,
  x: number, y: number, w: number, h: number,
  seed = 1,
  timeMs = 0,
): void {
  const p = paletteOf(skin);
  const t: TextureCtx = { ctx, x, y, w, h, rand: makeRand(seed * 2654435761) };
  switch (skin.texture) {
    case 'none': texNone(t, p.base, p.light); break;
    case 'wood': texWood(t, p.base, p.dark); break;
    case 'marble': texMarble(t, p.base, p.border); break;
    case 'brushed': texBrushed(t, p.base, p.dark); break;
    case 'ice': texIce(t, p.base, p.light); break;
    case 'neon': texNeon(t, p.base, p.glow); break;
    case 'gold': texGold(t, p.base, p.light, p.dark); break;
    case 'stardust': texStardust(t, p.base, p.light, timeMs); break;
    case 'galaxy': texGalaxy(t, p.base, p.glow, p.dark, timeMs); break;
  }
}

import { paintSkin, paletteOf, type Skin } from './skins';

export function drawBoard(
  ctx: CanvasRenderingContext2D,
  level: Level,
  layout: BoardLayout,
  state: BoardRenderState,
  skin: Skin,
  now: number,
): void {
  const { cell, originX, originY } = layout;
  const gap = Math.max(2, Math.floor(cell * 0.06));
  const size = cell - gap * 2;
  const p = paletteOf(skin);

  for (let y = 0; y < level.h; y++) {
    for (let x = 0; x < level.w; x++) {
      const idx = y * level.w + x;
      const c = level.grid[y][x];
      let px = originX + x * cell + gap;
      let py = originY + y * cell + gap;

      if (state.shakeCell === idx && state.shakeAt > 0) {
        const elapsed = now - state.shakeAt;
        if (elapsed < 200) px += Math.sin(elapsed / 200 * Math.PI * 6) * 4;
      }

      switch (c.type) {
        case CellType.BLOCK: {
          ctx.fillStyle = p.dark;
          roundRect(ctx, px, py, size, size, Math.floor(size * 0.18));
          ctx.fill();
          ctx.strokeStyle = 'rgba(0,0,0,0.4)';
          ctx.lineWidth = 2;
          ctx.beginPath();
          ctx.moveTo(px + size * 0.25, py + size * 0.25);
          ctx.lineTo(px + size * 0.75, py + size * 0.75);
          ctx.moveTo(px + size * 0.75, py + size * 0.25);
          ctx.lineTo(px + size * 0.25, py + size * 0.75);
          ctx.stroke();
          break;
        }
        case CellType.BRICK: {
          const lit = state.lit.has(idx);
          // 未点亮：暗色；已点亮：皮肤纹理
          if (lit) {
            paintSkin(ctx, skin, px, py, size, size, idx + 1, now);
            ctx.strokeStyle = p.border;
            ctx.lineWidth = 1.5;
            roundRect(ctx, px, py, size, size, Math.floor(size * 0.18));
            ctx.stroke();
          } else {
            ctx.fillStyle = p.dark;
            roundRect(ctx, px, py, size, size, Math.floor(size * 0.18));
            ctx.fill();
          }
          break;
        }
        case CellType.ONEWAY: {
          ctx.fillStyle = p.dark;
          roundRect(ctx, px, py, size, size, Math.floor(size * 0.18));
          ctx.fill();
          drawArrow(ctx, px + size / 2, py + size / 2, size * 0.26, c.dir, p.glow);
          break;
        }
        case CellType.START: {
          ctx.fillStyle = '#5EA94C';
          ctx.beginPath();
          ctx.arc(px + size / 2, py + size / 2, size / 2, 0, Math.PI * 2);
          ctx.fill();
          ctx.fillStyle = '#FFFFFF';
          ctx.font = `bold ${Math.floor(size * 0.36)}px system-ui, sans-serif`;
          ctx.textAlign = 'center';
          ctx.textBaseline = 'middle';
          ctx.fillText('起', px + size / 2, py + size / 2);
          break;
        }
        case CellType.END: {
          ctx.fillStyle = '#E97E6A';
          ctx.beginPath();
          ctx.arc(px + size / 2, py + size / 2, size / 2, 0, Math.PI * 2);
          ctx.fill();
          ctx.fillStyle = '#FFFFFF';
          ctx.font = `bold ${Math.floor(size * 0.36)}px system-ui, sans-serif`;
          ctx.textAlign = 'center';
          ctx.textBaseline = 'middle';
          ctx.fillText('终', px + size / 2, py + size / 2);
          break;
        }
      }

      if (state.hint.has(idx)) {
        ctx.strokeStyle = '#F2C14E';
        ctx.lineWidth = 4;
        roundRect(ctx, px + 2, py + 2, size - 4, size - 4, Math.floor(size * 0.18));
        ctx.stroke();
      }
    }
  }
}

  st: SkinsPageState,
  now: number,

  ctx.fillRect(0, 0, DESIGN_W, DESIGN_H);

  ctx.fillText(`已解锁 ${st.unlocked.size} / ${SKINS.length}`, DESIGN_W / 2, 125);

  const buttons: Button[] = [];

  const gapX = 30, gapY = 30;

    ctx.textBaseline = 'top';

      ctx.fillStyle = COLORS.blue;
      ctx.font = '20px system-ui, sans-serif';
      ctx.fillText('使用中', x + cardW / 2, y + pv + 60);
    } else if (owned) {
      ctx.fillStyle = COLORS.subText;
      ctx.font = '20px system-ui, sans-serif';
      ctx.fillText('点击装备', x + cardW / 2, y + pv + 60);
    } else {

      ctx.font = '20px system-ui, sans-serif';
      ctx.fillText(unlockLabel(skin), x + cardW / 2, y + pv + 60);
    }

    // 未解锁的遮罩

      ctx.fillStyle = 'rgba(10,12,20,0.5)';
      roundRect(ctx, x, y, cardW, cardH, 16);
      ctx.fill();
      ctx.fillStyle = COLORS.text;
      ctx.font = '36px system-ui, sans-serif';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText('🔒', x + cardW / 2, y + cardH / 2);
    }
  });

  // 底部：返回 / 翻页
  const backBtn: Button = { x: 40, y: DESIGN_H - 140, w: 160, h: 72, label: '返回', kind: 'secondary', id: 'back' };
  drawButton(ctx, backBtn, false);
  buttons.push(backBtn);

  const totalPages = Math.ceil(SKINS.length / PER_PAGE);
  if (st.page > 0) {
    const prev: Button = { x: DESIGN_W - 320, y: DESIGN_H - 140, w: 120, h: 72, label: '上一页', kind: 'ghost', id: 'prevPage' };
    drawButton(ctx, prev, false);
    buttons.push(prev);
  }
  if (st.page < totalPages - 1) {
    const next: Button = { x: DESIGN_W - 180, y: DESIGN_H - 140, w: 120, h: 72, label: '下一页', kind: 'ghost', id: 'nextPage' };
    drawButton(ctx, next, false);
    buttons.push(next);
  }

  return buttons;
}

/** 未解锁皮肤的提示文案。 */
function unlockLabel(skin: Skin): string {
  switch (skin.unlockType) {
    case 'level': return `通关第 ${skin.unlockValue} 关`;
    case 'milestone': return `通关 ${skin.unlockValue} 关`;
    case 'ad': return '看广告解锁';
    default: return '未开放';
  }
}

/** 判断点击落在哪个皮肤卡片上。 */
export function pickSkinCard(x: number, y: number, st: SkinsPageState): Skin | null {
  const cardW = 200, cardH = 240;
  const gapX = 30, gapY = 30;
  const gridW = cardW * 3 + gapX * 2;
  const startX = (DESIGN_W - gridW) / 2;
  const startY = 190;
  const visible = SKINS.slice(st.page * PER_PAGE, (st.page + 1) * PER_PAGE);

  for (let i = 0; i < visible.length; i++) {
    const col = i % 3, row = Math.floor(i / 3);
    const cx = startX + col * (cardW + gapX);
    const cy = startY + row * (cardH + gapY);
    if (x >= cx && x <= cx + cardW && y >= cy && y <= cy + cardH) {
      return visible[i];
    }
  }
  return null;
}

## 坑
- **`paintSkin` 每次渲染都重画纹理** → 星尘/星河有闪烁效果必须每帧重画；其它纹理每帧重画也 OK（格子数 ≤121）
- **纹理里的 `t.rand()` 必须是确定性的** → `makeRand(seed)` 用格子索引做种子，保证同一格每次纹理一致（否则闪烁噪点乱跳）
- **`ValueNoise` 的 perm 表**用 LCG 洗牌，确定性
- **未解锁遮罩会盖住解锁文案** → 遮罩用半透明 + 锁图标居中，条件文案放在卡片下方（当前实现在遮罩之前画，会被盖住）→ **调整顺序：先画遮罩再画文案，或把遮罩只画在预览区**
- **皮肤状态要同步服务端** → 皮肤列表在 `users.unlocked_skins`，装备的皮肤存本地即可（也可加服务端字段）
