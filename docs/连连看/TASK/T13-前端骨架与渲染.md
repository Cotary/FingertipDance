```ts
const mem = new Map<string, string>();

/**
 * Web 平台实现。广告用假倒计时模拟，登录返回 mock code。
 */
export const webPlatform: Platform = {
  name: 'web',

  async login() {
    return { code: 'web_mock_code_' + Date.now() };
  },

  async request<T>(url: string, method: 'GET' | 'POST', data?: unknown, header?: Record<string, string>): Promise<T> {
    const res = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json', ...(header ?? {}) },
      body: method === 'POST' ? JSON.stringify(data ?? {}) : undefined,
    });
    const json = (await res.json()) as { code: number; msg: string; data: T };
    if (json.code !== 0) throw new Error(`${json.code}: ${json.msg}`);
    return json.data;
  },

  storage: {
    get: (k) => (k in mem ? mem.get(k)! : localStorage.getItem(k)),
    set: (k, v) => { mem.set(k, v); try { localStorage.setItem(k, v); } catch { /* 隐私模式 */ } },
    remove: (k) => { mem.delete(k); try { localStorage.removeItem(k); } catch { /* noop */ } },

    async showRewarded(placement: string) {
      console.log('[mock ad] rewarded', placement);

      return true;

    console.log('[mock share]', opts);

    canvas.style.width = width + 'px';
    canvas.style.height = height + 'px';
    return { canvas, width, height, dpr };
```

```ts
    return new Promise((resolve, reject) => {
      wx.request({
        url,
        method,
        data,
        header: { 'content-type': 'application/json', ...(header ?? {}) },
        success: (res: any) => {
          const body = res.data as { code: number; msg: string; data: T };
          if (!body || body.code !== 0) {
            reject(new Error(`${body?.code ?? 'unknown'}: ${body?.msg ?? '请求失败'}`));
            return;
          }
          resolve(body.data);
        },
        fail: (err: any) => reject(new Error('网络请求失败: ' + err.errMsg)),
      });
    });
  },

  storage: {
    get: (k) => wx.getStorageSync(k) || null,
    set: (k, v) => wx.setStorageSync(k, v),
    remove: (k) => wx.removeStorageSync(k),
  },

  ad: {
    showRewarded(placement: string) {
      return new Promise<boolean>((resolve) => {
        const ad = wx.createRewardedVideoAd({ adUnitId: rewardedAdId });
        let settled = false;
        const finish = (ok: boolean) => {
          if (settled) return;
          settled = true;
          resolve(ok);
        };
        ad.onClose((res: any) => finish(!!res?.isEnded));
        ad.onError(() => { console.warn('广告加载失败', placement); finish(true); });  // 兜底发放
        ad.load()
          .then(() => ad.show())
          .catch(() => ad.load().then(() => ad.show()).catch(() => finish(true)));
      });
    },
    showInterstitial(placement: string) {
      try {
        const ad = wx.createInterstitialAd({ adUnitId: interstitialAdId });
        ad.onClose(() => {});
        ad.load().then(() => ad.show()).catch(() => { /* 不可用则静默跳过 */ });
      } catch {
        // 基础库不支持
      }
    },
    showBanner() {
      try {
        const ad = wx.createBannerAd({ adUnitId: bannerAdId, style: { left: 0, top: 0, width: 750 } });
        ad.onError(() => {});
        ad.load().then(() => ad.show()).catch(() => {});

    hideBanner() { /* 由 banner 引用统一管理，此处留空 */ },
  },

    wx.shareAppMessage({
      title: opts.title,
      imageUrl: opts.imageUrl,
      query: opts.query,
    });

  vibrate() {

    const width = sys.windowWidth;
    const height = sys.windowHeight;
    const canvas = wx.createCanvas();

    return { canvas, width, height, dpr };
```

```ts
/** 设计分辨率。 */
export const DESIGN_W = 750;
export const DESIGN_H = 1334;

/** 棋盘区域在设计坐标系里的边界。 */
export const BOARD_TOP = 220;
export const BOARD_BOTTOM = 1000;
export const BOARD_SIDE = 40;

/** 棋盘布局信息。 */
export interface BoardLayout {
  /** 每格像素尺寸（设计坐标系） */
  cell: number;
  /** 棋盘左上角（设计坐标系） */
  originX: number;
  originY: number;
  /** 棋盘宽高（设计坐标系） */
  boardW: number;
  boardH: number;
}

/**
 * 根据棋盘尺寸计算布局。
 */
export function computeLayout(w: number, h: number): BoardLayout {
  const maxW = DESIGN_W - BOARD_SIDE * 2;
  const maxH = BOARD_BOTTOM - BOARD_TOP;
  const cell = Math.floor(Math.min(maxW / w, maxH / h));
  const boardW = cell * w;
  const boardH = cell * h;
  return {
    cell,
    originX: Math.round((DESIGN_W - boardW) / 2),
    originY: Math.round(BOARD_TOP + (maxH - boardH) / 2),
    boardW,
    boardH,
  };
}

/** 把设计坐标系的点转成画布像素坐标。 */
export function toCanvas(scale: number, offsetX: number, offsetY: number, x: number, y: number): { x: number; y: number } {
  return { x: x * scale + offsetX, y: y * scale + offsetY };
}
```

### 13.6 写 `src/render/board.ts`（棋盘绘制）

```ts
import type { Level } from '../core/types';
import { CellType } from '../core/types';
import type { BoardLayout } from './layout';

export interface BoardRenderState {
  /** 已点亮的格子索引集合 */
  lit: Set<number>;
  /** 当前路径（格子索引序列） */
  path: number[];
  /** 提示高亮的格子 */
  hint: Set<number>;
  /** 断线抖动的格子索引，-1 表示无 */
  shakeCell: number;
  /** 抖动时间戳（ms），0 表示不抖 */
  shakeAt: number;
}

/**
 * 绘制棋盘与所有格子。
 */
export function drawBoard(
  ctx: CanvasRenderingContext2D,
  level: Level,
  layout: BoardLayout,
  state: BoardRenderState,
  palette: { block: string; brick: string; brickLit: string; oneway: string; start: string; end: string },
  now: number,
): void {
  const { cell, originX, originY } = layout;
  const gap = Math.max(2, Math.floor(cell * 0.06));
  const size = cell - gap * 2;

  for (let y = 0; y < level.h; y++) {
    for (let x = 0; x < level.w; x++) {
      const idx = y * level.w + x;
      const cellData = level.grid[y][x];
      let px = originX + x * cell + gap;
      let py = originY + y * cell + gap;

      // 断线抖动
      if (state.shakeCell === idx && state.shakeAt > 0) {
        const elapsed = now - state.shakeAt;
        if (elapsed < 200) {
          px += Math.sin(elapsed / 200 * Math.PI * 6) * 4;
        }
      }

      drawCell(ctx, cellData.type, cellData.dir, px, py, size, state.lit.has(idx), palette, state.hint.has(idx));
    }
  }
}

/** 绘制单个格子。 */
function drawCell(
  ctx: CanvasRenderingContext2D,
  type: number,
  dir: number,
  x: number, y: number, size: number,
  lit: boolean,
  palette: { block: string; brick: string; brickLit: string; oneway: string; start: string; end: string },
  hinted: boolean,
): void {
  const r = Math.floor(size * 0.18);

  ctx.save();
  switch (type) {
    case CellType.BLOCK: {
      ctx.fillStyle = palette.block;
      roundRect(ctx, x, y, size, size, r);
      ctx.fill();
      // 对角交叉线
      ctx.strokeStyle = 'rgba(0,0,0,0.35)';
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.moveTo(x + size * 0.25, y + size * 0.25);
      ctx.lineTo(x + size * 0.75, y + size * 0.75);
      ctx.moveTo(x + size * 0.75, y + size * 0.25);
      ctx.lineTo(x + size * 0.25, y + size * 0.75);
      ctx.stroke();
      break;
    }
    case CellType.BRICK: {
      ctx.fillStyle = lit ? palette.brickLit : palette.brick;
      roundRect(ctx, x, y, size, size, r);
      ctx.fill();
      if (lit) {
        // 顶部高光
        ctx.fillStyle = 'rgba(255,255,255,0.25)';
        roundRect(ctx, x + size * 0.12, y + size * 0.1, size * 0.76, size * 0.22, r * 0.5);
        ctx.fill();
      }
      break;
    }
    case CellType.ONEWAY: {
      ctx.fillStyle = palette.oneway;
      roundRect(ctx, x, y, size, size, r);
      ctx.fill();
      drawArrow(ctx, x + size / 2, y + size / 2, size * 0.3, dir, '#FFFFFF');
      break;
    }
    case CellType.START: {
      ctx.fillStyle = palette.start;
      ctx.beginPath();
      ctx.arc(x + size / 2, y + size / 2, size / 2, 0, Math.PI * 2);
      ctx.fill();
      break;
    }
    case CellType.END: {
      ctx.fillStyle = palette.end;
      ctx.beginPath();
      ctx.arc(x + size / 2, y + size / 2, size / 2, 0, Math.PI * 2);
      ctx.fill();
      break;
    }
  }

  if (hinted) {
    ctx.strokeStyle = '#F2C14E';
    ctx.lineWidth = 4;
    roundRect(ctx, x + 2, y + 2, size - 4, size - 4, r);
    ctx.stroke();
  }
  ctx.restore();
}

/** 绘制箭头。dir: 0=上 1=下 2=左 3=右 */
export function drawArrow(ctx: CanvasRenderingContext2D, cx: number, cy: number, r: number, dir: number, color: string): void {
  ctx.save();
  ctx.translate(cx, cy);
  ctx.rotate(dir * Math.PI / 2);
  ctx.fillStyle = color;
  ctx.beginPath();
  ctx.moveTo(0, -r);
  ctx.lineTo(r * 0.72, r * 0.5);
  ctx.lineTo(-r * 0.72, r * 0.5);
  ctx.closePath();
  ctx.fill();
  ctx.restore();
}

/** 画圆角矩形路径。 */
export function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
  const rr = Math.min(r, w / 2, h / 2);
  ctx.beginPath();
  ctx.moveTo(x + rr, y);
  ctx.arcTo(x + w, y, x + w, y + h, rr);
  ctx.arcTo(x + w, y + h, x, y + h, rr);
  ctx.arcTo(x, y + h, x, y, rr);
  ctx.arcTo(x, y, x + w, y, rr);
  ctx.closePath();
}
```

```ts
import type { BoardLayout } from './layout';

/**
 * 绘制玩家路径连线。
 */
export function drawTrail(
  ctx: CanvasRenderingContext2D,
  path: number[],
  w: number,
  layout: BoardLayout,
  color: string,
  glow: string,
): void {
  if (path.length === 0) return;
  const { cell, originX, originY } = layout;

  const pts = path.map((idx) => ({
    x: originX + (idx % w) * cell + cell / 2,
    y: originY + Math.floor(idx / w) * cell + cell / 2,
  }));

  ctx.save();
  ctx.lineCap = 'round';
  ctx.lineJoin = 'round';

  // 外发光
  ctx.strokeStyle = glow;
  ctx.lineWidth = cell * 0.22;
  ctx.shadowColor = glow;
  ctx.shadowBlur = 12;
  ctx.beginPath();
  pts.forEach((p, i) => { if (i === 0) ctx.moveTo(p.x, p.y); else ctx.lineTo(p.x, p.y); });
  ctx.stroke();

  // 主线
  ctx.shadowBlur = 0;
  ctx.strokeStyle = color;
  ctx.lineWidth = cell * 0.14;
  ctx.beginPath();
  pts.forEach((p, i) => { if (i === 0) ctx.moveTo(p.x, p.y); else ctx.lineTo(p.x, p.y); });
  ctx.stroke();

  // 白色内芯
  ctx.strokeStyle = 'rgba(255,255,255,0.55)';
  ctx.lineWidth = cell * 0.05;
  ctx.beginPath();
  pts.forEach((p, i) => { if (i === 0) ctx.moveTo(p.x, p.y); else ctx.lineTo(p.x, p.y); });
  ctx.stroke();

  ctx.restore();
}
```

### 13.9 写 `src/game/state.ts`（关内状态机）

```ts
import { CellType, type Level } from '../core/types';
import { idxOf, canEnter } from '../core/grid';

export type Phase = 'idle' | 'dragging' | 'won' | 'lost';

/** 关内状态。 */
export interface GameState {
  level: Level;
  phase: Phase;
  /** 当前路径（格子索引序列） */
  path: number[];
  /** 已点亮的格子 */
  lit: Set<number>;
  /** 提示高亮 */
  hint: Set<number>;
  /** 断线抖动的格子 */
  shakeCell: number;
  /** 断线时间戳 */
  shakeAt: number;
  /** 本关开始时间 */
  startAt: number;
  /** 已用提示次数 */
  hintUsed: number;
  /** 是否被跳过 */
  skipped: boolean;
}

/** 创建初始状态。 */
export function createState(level: Level): GameState {
  return {
    level,
    phase: 'idle',
    path: [],
    lit: new Set(),
    hint: new Set(),
    shakeCell: -1,
    shakeAt: 0,
    startAt: Date.now(),
    hintUsed: 0,
    skipped: false,
  };
}

/** 找到起点索引。 */
export function findStart(level: Level): number {
  for (let y = 0; y < level.h; y++) {
    for (let x = 0; x < level.w; x++) {
      if (level.grid[y][x].type === CellType.START) return idxOf(x, y, level.w);
    }
  }
  return -1;
}

/**
 * 尝试移动到指定格子。
 * @returns 'moved' | 'break' | 'blocked' | 'same'
 */
export function tryMove(state: GameState, targetIdx: number): 'moved' | 'break' | 'blocked' | 'same' {
  if (state.phase !== 'dragging') return 'blocked';
  if (state.path.length === 0) return 'blocked';

  const last = state.path[state.path.length - 1];
  if (targetIdx === last) return 'same';

  const w = state.level.w;
  const lx = last % w, ly = Math.floor(last / w);
  const tx = targetIdx % w, ty = Math.floor(targetIdx / w);
  const dx = tx - lx, dy = ty - ly;

  // 非相邻 → 断线
  if (Math.abs(dx) + Math.abs(dy) !== 1) {
    state.phase = 'idle';
    state.shakeCell = last;
    state.shakeAt = Date.now();
    return 'break';
  }

  // 目标不可进入
  const enterDir = dirOf(dx, dy);
  if (!canEnter(state.level.grid, last, targetIdx, enterDir)) return 'blocked';

  state.path.push(targetIdx);
  state.lit.add(targetIdx);
  return 'moved';
}

/** 由坐标差求方向。 */
export function dirOf(dx: number, dy: number): 0 | 1 | 2 | 3 {
  if (dy === -1) return 0;
  if (dy === 1) return 1;
  if (dx === -1) return 2;
  return 3;
}

/** 松手：判断是否通关。 */
export function release(state: GameState): boolean {
  if (state.phase !== 'dragging') return false;
  state.phase = 'idle';
  const last = state.path[state.path.length - 1];
  const cell = state.level.grid[Math.floor(last / state.level.w)][last % state.level.w];
  if (cell.type === CellType.END) {
    state.phase = 'won';
    return true;
  }
  // 没到终点 → 本段作废
  state.path = [];
  state.lit.clear();
  return false;
}
```

### 13.10 写 `src/main.ts`（主循环 + 渲染）

```ts
import { platform } from './platform';
import { login, fetchConfig, fetchLevels } from './net/api';
import { createState, findStart, tryMove, release, type GameState } from './game/state';
import { computeLayout, DESIGN_W, DESIGN_H, type BoardLayout } from './render/layout';
import { drawBoard } from './render/board';
import { drawTrail } from './render/trail';
import { decodeGrid } from './core/codec';
import { flatten } from './core/grid';
import { CellType, type Level } from './core/types';

const PALETTE = {
  block: '#39415C',
  brick: '#5A6480',
  brickLit: '#6692F5',
  oneway: '#9B7BE8',
  start: '#5EA94C',
  end: '#E97E6A',
};

const { canvas, width, height, dpr } = platform.getCanvas();
const ctx = canvas.getContext('2d')!;

/** 缩放与偏移。 */
const scale = Math.min(width / DESIGN_W, height / DESIGN_H);
const offsetX = (width - DESIGN_W * scale) / 2;
const offsetY = (height - DESIGN_H * scale) / 2;

let state: GameState | null = null;
let layout: BoardLayout | null = null;
let pointerDown = false;

/** 画布坐标 → 设计坐标系坐标。 */
function toDesign(clientX: number, clientY: number): { x: number; y: number } {
  const rect = canvas.getBoundingClientRect();
  const px = (clientX - rect.left) * (canvas.width / rect.width);
  const py = (clientY - rect.top) * (canvas.height / rect.height);
  return { x: (px / dpr - offsetX) / scale, y: (py / dpr - offsetY) / scale };
}

/** 设计坐标 → 最近的格子索引。 */
function pickCell(dx: number, dy: number): number {
  if (!state || !layout) return -1;
  const { cell, originX, originY } = layout;
  const x = Math.floor((dx - originX) / cell);
  const y = Math.floor((dy - originY) / cell);
  if (x < 0 || y < 0 || x >= state.level.w || y >= state.level.h) return -1;
  return y * state.level.w + x;
}

/** 触摸事件处理。 */
function onDown(clientX: number, clientY: number) {
  if (!state || state.phase !== 'idle') return;
  const d = toDesign(clientX, clientY);
  const idx = pickCell(d.x, d.y);
  if (idx !== findStart(state.level)) return;
  state.phase = 'dragging';
  state.path = [idx];
  state.lit.add(idx);
  pointerDown = true;
}

function onMove(clientX: number, clientY: number) {
  if (!state || !pointerDown) return;
  const d = toDesign(clientX, clientY);
  const idx = pickCell(d.x, d.y);
  if (idx < 0) return;
  const result = tryMove(state, idx);
  if (result === 'break') {
    platform.vibrate();
    pointerDown = false;
  }
}

function onUp() {
  if (!state || !pointerDown) return;
  pointerDown = false;
  const won = release(state);
  if (won) console.log('通关！步数', state.path.length - 1);
}

// Web 端事件绑定
if (platform.name === 'web') {
  canvas.addEventListener('pointerdown', (e) => onDown(e.clientX, e.clientY));
  canvas.addEventListener('pointermove', (e) => onMove(e.clientX, e.clientY));
  canvas.addEventListener('pointerup', onUp);
  canvas.addEventListener('pointercancel', onUp);
}

/** 渲染循环。 */
function frame() {
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  ctx.scale(dpr * scale, dpr * scale);
  ctx.translate(offsetX / dpr, offsetY / dpr);

  // 背景
  ctx.fillStyle = '#141824';
  ctx.fillRect(0, 0, DESIGN_W, DESIGN_H);

  if (state && layout) {
    drawBoard(ctx, state.level, layout, {
      lit: state.lit,
      path: state.path,
      hint: state.hint,
      shakeCell: state.shakeCell,
      shakeAt: state.shakeAt,
    }, PALETTE, Date.now());

    drawTrail(ctx, state.path, state.level.w, layout, '#6692F5', 'rgba(102,146,245,0.35)');
  }

  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

/** 启动流程。 */
async function boot() {
  try {
    await login();
    const cfg = await fetchConfig();
    const levels = await fetchLevels(1, 20);
    if (levels.length === 0) throw new Error('关卡表为空');
    state = createState(levels[0]);
    layout = computeLayout(levels[0].w, levels[0].h);
    console.log('启动成功，关卡版本', cfg.levelVersion);
  } catch (e) {
    console.error('启动失败', e);
  }
}
boot();
```

## 坑
- **`decodeGrid` 用到 `atob`** → Web 端有，Node 端需要兜底（已在 T02 处理）
- `platform.getCanvas()` 在微信端调 `wx.createCanvas()` → **必须只调一次**，重复调用会创建多个画布
- `canvas.getBoundingClientRect()` 在微信端不存在 → `toDesign` 需要平台分支：
  ```ts
  function toDesign(clientX: number, clientY: number) {
    if (platform.name === 'wechat') {
      // 微信触摸事件给的是逻辑像素，直接用
      return { x: (clientX - offsetX) / scale, y: (clientY - offsetY) / scale };
    }
    const rect = canvas.getBoundingClientRect();
    ...
  }
  ```
  **必须在 T13 就做这个分支**，否则 T21 移植时会踩坑
- `requestAnimationFrame` 在微信小游戏可用 → OK
- `state.path` 在断线后保留（按 GDD 3.2）→ 但 `tryMove` 里断线时 `phase='idle'` 且 `path` 未清空 → **需要在断线时把 path 回退到上一个合法点**。当前实现是清空整个 path，与 GDD 不符。**修正**：
  ```ts
  if (Math.abs(dx) + Math.abs(dy) !== 1) {
    // 断线：回退到上一个合法点（保留已点亮砖块）
    if (state.path.length > 1) state.path.pop();
    state.phase = 'idle';
    state.shakeCell = last;
    state.shakeAt = Date.now();
    return 'break';
  }
  ```
  > **GDD 3.2 说"线头回退到上一个合法点"，但如果玩家继续按着拖，是否应该续上？**
  > **决策**：断线后 `phase = 'idle'`，玩家必须松手重新从起点开始。**这与 GDD 的"断线回退"不一致。**
  > **需要在 GDD 里明确**：建议改为「断线只回退一格，玩家可继续拖动」→ `phase` 保持 `'dragging'`，path 弹出最后一格。这样体验更连续。
  > **执行时**：把 `tryMove` 断线分支改为：
  > ```ts
  > if (Math.abs(dx) + Math.abs(dy) !== 1) {
  >   if (state.path.length > 1) state.path.pop();
  >   state.shakeCell = last;
  >   state.shakeAt = Date.now();
  >   return 'break';   // phase 仍为 dragging，玩家可继续
  > }
  > ```
  > 并在 `01-GDD.md` 3.2 补一句：「断线只回退一格，玩家可继续拖动；断线不扣分不失败」。
