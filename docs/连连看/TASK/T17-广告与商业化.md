### 17.1 写 `src/ads/ads.ts`（广告管理器）

```ts
import { platform } from '../platform';
import { reportEvent } from '../net/api';

/** 广告位。 */
export type Placement = 'hint' | 'skip' | 'skin_unlock' | 'result_interstitial' | 'skin_banner';

/**
 * 广告管理器。统一处理奖励判定、失败兜底与埋点。
 */
export class Ads {
  /** 上次插屏时间戳。 */
  private lastInterstitialAt = 0;
  /** 插屏最小间隔（毫秒）。 */
  private interstitialInterval = 60_000;

  /**
   * 展示激励视频。
   * @param placement 广告位，用于埋点
   * @returns true = 完整看完（或加载失败兜底发放），false = 中途退出
   */
  async showRewarded(placement: Placement): Promise<boolean> {
    reportEvent('ad_start', { ad_type: 'rewarded', placement });
    try {
      const completed = await platform.ad.showRewarded(placement);
      if (completed) {
        reportEvent('ad_complete', { ad_type: 'rewarded', placement });
      } else {
        reportEvent('ad_abort', { ad_type: 'rewarded', placement });
      }
      return completed;
    } catch (e) {
      reportEvent('ad_error', { ad_type: 'rewarded', placement, err_msg: String(e) });
      // 兜底：广告不可用时直接发放，避免玩家点击无响应
      return true;
    }
  }

  /**
   * 展示插屏。带频控。
   */
  async showInterstitial(placement: Placement): Promise<void> {
    const now = Date.now();
    if (now - this.lastInterstitialAt < this.interstitialInterval) return;
    this.lastInterstitialAt = now;
    reportEvent('ad_start', { ad_type: 'interstitial', placement });
    try {
      await platform.ad.showInterstitial(placement);
      reportEvent('ad_complete', { ad_type: 'interstitial', placement });
    } catch {
      reportEvent('ad_error', { ad_type: 'interstitial', placement });
    }
  }

  /** 显示 Banner。 */
  showBanner(): void {
    platform.ad.showBanner();
  }

  /** 隐藏 Banner。 */
  hideBanner(): void {
    platform.ad.hideBanner();
  }
}

/** 全局单例。 */
export const ads = new Ads();
```

### 17.2 写 `src/ads/rewardRules.ts`（奖励规则）

```ts
import { platform } from '../platform';

/** 提示的免费次数记录键。 */
const KEY_FREE_HINT = 'free_hint_used';

/**
 * 检查本关提示是否需要看广告。
 * 规则：每关第一次提示必须看广告，之后免费。
 */
export function needsAdForHint(levelIdx: number, hintUsed: number): boolean {
  return hintUsed === 0;
}

/**
 * 检查本关跳过是否超过每日限额。
 * 客户端做本地预检（服务端也会校验）。
 */
export function canSkipToday(levelIdx: number): boolean {
  const day = new Date().toISOString().slice(0, 10);
  const key = `skip_${day}_${levelIdx}`;
  return platform.storage.get(key) !== '1';
}

/**
 * 标记本关今日已跳过。
 */
export function markSkippedToday(levelIdx: number): void {
  const day = new Date().toISOString().slice(0, 10);
  platform.storage.set(`skip_${day}_${levelIdx}`, '1');
}

/**
 * 检查皮肤是否已解锁过（防止重复看广告）。
 */
export function isSkinUnlocked(unlocked: Set<string>, skinId: string): boolean {
  return unlocked.has(skinId);
}
```

### 17.3 在 `src/main.ts` 接入广告

```ts
import { ads } from './ads/ads';
import { needsAdForHint, canSkipToday, markSkippedToday } from './ads/rewardRules';

// 使用提示
async function useHint() {
  const st = session.state;
  if (!st) return;

  const head = st.path.length ? st.path[st.path.length - 1] : findStartIdx(st);
  const hint = findHint(st.level.grid, { x: head % st.level.w, y: Math.floor(head / st.level.w) }, 2);
  if (hint.length === 0) {
    console.log('当前无法给出提示');
    return;
  }

  // 首次使用需看广告
  let viaAd = false;
  if (needsAdForHint(st.level.idx, st.hintUsed)) {
    const ok = await ads.showRewarded('hint');
    if (!ok) return;   // 中途退出 → 不给提示
    viaAd = true;
  }

  session.setHint(hint);
  reportEvent('hint_use', { level_idx: st.level.idx, via_ad: viaAd });
}

// 跳过
async function useSkip() {
  const st = session.state;
  if (!st) return;
  if (!canSkipToday(st.level.idx)) {
    console.log('本关今天已跳过');
    return;
  }
  const ok = await ads.showRewarded('skip');
  if (!ok) return;

  markSkippedToday(st.level.idx);
  await reportSkip(st.level.idx);
  session.markSkipped();
  reportEvent('skip_use', { level_idx: st.level.idx, via_ad: true });
  void onWin(st.level.shortest, 0, [], true);
}

// 皮肤广告解锁
async function unlockByAd(skin: Skin) {
  const ok = await ads.showRewarded('skin_unlock');
  if (!ok) return;
  progress.unlockSkin(skin.id);
  skinsPage.unlocked.add(skin.id);
  skinsPage.equipped = skin.id;
  progress.equipSkin(skin.id);
  audio.unlock();
  reportEvent('skin_unlock', { skin_id: skin.id, via: 'ad' });
}
```

## 坑
- **广告加载失败必须兜底发放** → `Ads.showRewarded` 的 catch 里 return true
- **中途退出不发奖但也不批评** → 只记 `ad_abort` 埋点，不弹任何提示
- **插屏频控** → 60 秒间隔；`lastInterstitialAt` 初值 0 → 第一次调用时 `now - 0` 很大，能通过
- **跳过的关卡不发皮肤** → 服务端 `applyProgress` 里 `if !req.Skipped && ...`；客户端 `reportSkip` 走的是 `/api/skip`，该接口也不发皮肤
- **跳过的关卡 `max_level` 仍推进** → 否则玩家卡死
- **Banner 不能挡主按钮** → 皮肤页底部按钮在 `DESIGN_H - 140`，Banner 高度 80 → 需要把按钮上移或 Banner 放更底。**建议**：Banner 在微信端用 `style.top` 定位到屏幕最底部，Canvas 设计坐标 1334 对应实际高度，Banner 会覆盖最底部 80px → **把皮肤页按钮 y 改成 `DESIGN_H - 220`**
