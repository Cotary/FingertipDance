### 14.1 写 `src/game/session.ts`（关卡会话管理）

```ts
import type { Level } from '../core/types';
import { createState, findStart, tryMove, release, type GameState } from './state';
import { computeLayout, type BoardLayout } from '../render/layout';

export type SessionListener = (e: SessionEvent) => void;

export type SessionEvent =
  | { type: 'level_loaded'; level: Level }
  | { type: 'moved'; idx: number }
  | { type: 'break' }
  | { type: 'blocked' }
  | { type: 'won'; steps: number; timeMs: number; path: number[] }
  | { type: 'abandoned' };

/**
 * 关卡会话。管理当前关卡的状态、输入与事件广播。
 */
export class Session {
  state: GameState | null = null;
  layout: BoardLayout | null = null;

  private listeners: SessionListener[] = [];
  private dragging = false;

  /** 订阅事件。 */
  on(fn: SessionListener): void {
    this.listeners.push(fn);
  }

  /** 广播事件。 */
  private emit(e: SessionEvent): void {
    for (const fn of this.listeners) fn(e);
  }

  /** 载入关卡。 */
  load(level: Level): void {
    this.state = createState(level);
    this.layout = computeLayout(level.w, level.h);
    this.dragging = false;
    this.emit({ type: 'level_loaded', level });
  }

  /** 按下。 */
  pointerDown(x: number, y: number): void {
    if (!this.state || this.state.phase !== 'idle') return;
    const idx = this.pick(x, y);
    if (idx !== findStart(this.state.level)) return;
    this.state.phase = 'dragging';
    this.state.path = [idx];
    this.state.lit.add(idx);
    this.dragging = true;
  }

  /** 移动。 */
  pointerMove(x: number, y: number): void {
    if (!this.state || !this.dragging) return;
    const idx = this.pick(x, y);
    if (idx < 0) return;
    const r = tryMove(this.state, idx);
    if (r === 'moved') this.emit({ type: 'moved', idx });
    else if (r === 'break') { this.emit({ type: 'break' }); this.dragging = false; }
    else if (r === 'blocked') this.emit({ type: 'blocked' });
  }

  /** 松手。 */
  pointerUp(): void {
    if (!this.state || !this.dragging) return;
    this.dragging = false;
    const won = release(this.state);
    if (won && this.state) {
      this.emit({
        type: 'won',
        steps: this.state.path.length - 1,
        timeMs: Date.now() - this.state.startAt,
        path: this.state.path.slice(),
      });
    }
  }

  /** 提示高亮。 */
  setHint(cells: number[]): void {
    if (!this.state) return;
    this.state.hint = new Set(cells);
    this.state.hintUsed++;
  }

  /** 标记为跳过通关。 */
  markSkipped(): void {
    if (!this.state) return;
    this.state.skipped = true;
    this.state.phase = 'won';
  }

  /** 放弃当前关卡（退出游戏时调用）。 */
  abandon(): void {
    if (!this.state || this.state.phase === 'won') return;
    this.emit({ type: 'abandoned' });
  }

  /** 设计坐标 → 格子索引。 */
  private pick(x: number, y: number): number {
    if (!this.state || !this.layout) return -1;
    const { cell, originX, originY } = this.layout;
    const cx = Math.floor((x - originX) / cell);
    const cy = Math.floor((y - originY) / cell);
    if (cx < 0 || cy < 0 || cx >= this.state.level.w || cy >= this.state.level.h) return -1;
    return cy * this.state.level.w + cx;
  }
}
```

### 14.2 写 `src/game/progressStore.ts`（进度与关卡缓存）

```ts
import { platform } from '../platform';
import { fetchLevels } from '../net/api';
import type { Level } from '../core/types';

/** 本地缓存键。 */
const KEY_LEVELS = 'levels_cache';
const KEY_PROGRESS = 'progress';

/** 缓存的关卡表。 */
interface CachedLevels {
  version: string;
  levels: Level[];
}

/** 玩家进度。 */
export interface Progress {
  maxLevel: number;
  unlockedSkins: string[];
  currentSkin: string;
  adWatchCnt: number;
}

/**
 * 关卡表缓存。版本不一致时自动重拉。
 */
export class LevelStore {
  private levels: Level[] = [];
  private version = '';

  /** 全部已加载的关卡。 */
  get all(): Level[] { return this.levels; }

  /** 某关（未加载返回 null）。 */
  get(idx: number): Level | null {
    return this.levels.find((l) => l.idx === idx) ?? null;
  }

  /** 当前最高已加载关号。 */
  get loadedMax(): number {
    return this.levels.length ? this.levels[this.levels.length - 1].idx : 0;
  }

  /**
   * 初始化：优先用缓存，版本不符则重拉。
   */
  async init(serverVersion: string): Promise<void> {
    const raw = platform.storage.get(KEY_LEVELS);
    if (raw) {
      try {
        const cached = JSON.parse(raw) as CachedLevels;
        if (cached.version === serverVersion && cached.levels.length > 0) {
          this.levels = cached.levels;
          this.version = cached.version;
          return;
        }
      } catch {
        // 缓存损坏 → 重拉
      }
    }
    await this.loadMore(serverVersion, 1, 50);
  }

  /**
   * 加载更多关卡。
   */
  async loadMore(serverVersion: string, from: number, count: number): Promise<void> {
    const fresh = await fetchLevels(from, count);
    if (fresh.length === 0) return;
    const byIdx = new Map(this.levels.map((l) => [l.idx, l]));
    for (const l of fresh) byIdx.set(l.idx, l);
    this.levels = Array.from(byIdx.values()).sort((a, b) => a.idx - b.idx);
    this.version = serverVersion;
    platform.storage.set(KEY_LEVELS, JSON.stringify({ version: serverVersion, levels: this.levels }));
  }

  /**
   * 确保某关已加载，不足则拉取。
   */
  async ensure(idx: number, serverVersion: string): Promise<Level | null> {
    let lv = this.get(idx);
    if (lv) return lv;
    const from = this.loadedMax > 0 ? this.loadedMax + 1 : idx;
    await this.loadMore(serverVersion, from, 50);
    lv = this.get(idx);
    return lv;
  }
}

/**
 * 进度存取。服务端为准，本地做乐观缓存。
 */
export class ProgressStore {
  private progress: Progress = {
    maxLevel: 1,
    unlockedSkins: ['wood', 'mint'],
    currentSkin: 'wood',
    adWatchCnt: 0,
  };

  /** 读取进度。 */
  get current(): Progress { return this.progress; }

  /** 从服务端同步。 */
  sync(p: Partial<Progress>): void {
    this.progress = { ...this.progress, ...p };
    platform.storage.set(KEY_PROGRESS, JSON.stringify(this.progress));
  }

  /** 从本地缓存恢复（启动时先渲染，再等服务端）。 */
  restore(): void {
    const raw = platform.storage.get(KEY_PROGRESS);
    if (!raw) return;
    try {
      this.progress = { ...this.progress, ...(JSON.parse(raw) as Partial<Progress>) };
    } catch { /* 忽略损坏缓存 */ }
  }

  /** 解锁皮肤。 */
  unlockSkin(id: string): void {
    if (!this.progress.unlockedSkins.includes(id)) {
      this.progress.unlockedSkins.push(id);
    }
    platform.storage.set(KEY_PROGRESS, JSON.stringify(this.progress));
  }

  /** 装备皮肤。 */
  equipSkin(id: string): void {
    this.progress.currentSkin = id;
    platform.storage.set(KEY_PROGRESS, JSON.stringify(this.progress));
  }
}
```

### 14.3 写 `src/ui/hud.ts`（HUD 与结算页，全部 Canvas 绘制）

```ts
import { DESIGN_W, DESIGN_H } from '../render/layout';
import { roundRect } from '../render/board';

export const COLORS = {
  bg: '#141824',
  bgLight: '#1E2436',
  panel: '#262D42',
  border: '#39415C',
  text: '#E8ECF5',
  subText: '#8A94AD',
  accent: '#E97E6A',
  success: '#5EA94C',
  blue: '#6692F5',
  yellow: '#F2C14E',
  purple: '#9B7BE8',
};

/** 矩形按钮。 */
export interface Button {
  x: number; y: number; w: number; h: number;
  label: string;
  kind: 'primary' | 'secondary' | 'ghost';
  id: string;
}

/**
 * 画一个按钮并返回它是否被点击。
 */
export function drawButton(ctx: CanvasRenderingContext2D, b: Button, pressed: boolean): void {
  const scale = pressed ? 0.96 : 1;
  const cx = b.x + b.w / 2, cy = b.y + b.h / 2;
  ctx.save();
  ctx.translate(cx, cy);
  ctx.scale(scale, scale);
  ctx.translate(-cx, -cy);

  if (b.kind === 'primary') {
    ctx.fillStyle = COLORS.blue;
    roundRect(ctx, b.x, b.y, b.w, b.h, 24);
    ctx.fill();
    ctx.fillStyle = '#FFFFFF';
  } else if (b.kind === 'secondary') {
    ctx.strokeStyle = COLORS.blue;
    ctx.lineWidth = 2;
    roundRect(ctx, b.x, b.y, b.w, b.h, 24);
    ctx.stroke();
    ctx.fillStyle = COLORS.blue;
  } else {
    ctx.fillStyle = 'rgba(255,255,255,0.08)';
    roundRect(ctx, b.x, b.y, b.w, b.h, 24);
    ctx.fill();
    ctx.fillStyle = COLORS.text;
  }

  ctx.font = '600 32px system-ui, sans-serif';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText(b.label, cx, cy);
  ctx.restore();
}

/**
 * 判断点是否落在按钮内。
 */
export function hitButton(b: Button, x: number, y: number): boolean {
  return x >= b.x && x <= b.x + b.w && y >= b.y && y <= b.y + b.h;
}

/** 画 HUD 顶部。 */
export function drawHud(ctx: CanvasRenderingContext2D, levelIdx: number, skinName: string, skinCount: number, totalSkins: number): void {
  ctx.fillStyle = COLORS.text;
  ctx.font = '600 34px system-ui, sans-serif';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'top';
  ctx.fillText(`第 ${levelIdx} 关`, DESIGN_W / 2, 110);

  ctx.fillStyle = COLORS.subText;
  ctx.font = '24px system-ui, sans-serif';
  ctx.textAlign = 'left';
  ctx.fillText(`皮肤 ${skinCount}/${totalSkins}`, 40, 160);
  ctx.textAlign = 'right';
  ctx.fillText(skinName, DESIGN_W - 40, 160);
}

/** 画进度条。 */
export function drawProgressBar(ctx: CanvasRenderingContext2D, current: number, total: number): void {
  const x = 40, y = 1040, w = DESIGN_W - 80, h = 16;
  ctx.fillStyle = COLORS.border;
  roundRect(ctx, x, y, w, h, 8);
  ctx.fill();

  const ratio = total > 0 ? Math.min(1, current / total) : 0;
  const fillW = Math.max(0, w * ratio);
  if (fillW > 0) {
    ctx.fillStyle = COLORS.blue;
    roundRect(ctx, x, y, fillW, h, 8);
    ctx.fill();
  }

  ctx.fillStyle = COLORS.subText;
  ctx.font = '22px system-ui, sans-serif';
  ctx.textAlign = 'left';
  ctx.textBaseline = 'top';
  ctx.fillText(`${current} / ${total} 关`, x, y + 28);
}

/** 画通关结算页。 */
export function drawResult(
  ctx: CanvasRenderingContext2D,
  levelIdx: number,
  steps: number,
  shortest: number,
  timeMs: number,
  unlockedSkin: { id: string; name: string } | null,
  progressRatio: number,
): Button[] {
  // 遮罩
  ctx.fillStyle = 'rgba(10,12,20,0.82)';
  ctx.fillRect(0, 0, DESIGN_W, DESIGN_H);

  // 面板
  const px = 50, py = 380, pw = DESIGN_W - 100, ph = 560;
  ctx.fillStyle = COLORS.panel;
  roundRect(ctx, px, py, pw, ph, 24);
  ctx.fill();

  ctx.textAlign = 'center';
  ctx.textBaseline = 'top';
  ctx.fillStyle = COLORS.success;
  ctx.font = '700 44px system-ui, sans-serif';
  ctx.fillText(`第 ${levelIdx} 关 通关！`, DESIGN_W / 2, py + 40);

  const ratio = shortest > 0 ? steps / shortest : 1;
  const stars = ratio <= 1.15 ? 3 : ratio <= 1.5 ? 2 : 1;
  ctx.fillStyle = COLORS.yellow;
  ctx.font = '40px system-ui, sans-serif';
  ctx.fillText('★'.repeat(stars) + '☆'.repeat(3 - stars), DESIGN_W / 2, py + 110);

  ctx.fillStyle = COLORS.text;
  ctx.font = '26px system-ui, sans-serif';
  ctx.fillText(`步数 ${steps}（最短 ${shortest}）`, DESIGN_W / 2, py + 175);
  ctx.fillText(`用时 ${(timeMs / 1000).toFixed(1)} 秒`, DESIGN_W / 2, py + 215);

  if (unlockedSkin) {
    ctx.fillStyle = COLORS.purple;
    ctx.font = '600 28px system-ui, sans-serif';
    ctx.fillText(`解锁新皮肤：${unlockedSkin.name}`, DESIGN_W / 2, py + 275);
  }

  ctx.fillStyle = COLORS.subText;
  ctx.font = '22px system-ui, sans-serif';
  ctx.fillText(`进度 ${(progressRatio * 100).toFixed(1)}%`, DESIGN_W / 2, py + 330);

  const nextBtn: Button = { x: 120, y: py + 400, w: DESIGN_W - 240, h: 88, label: '下一关', kind: 'primary', id: 'next' };
  const shareBtn: Button = { x: 120, y: py + 500, w: (DESIGN_W - 260) / 2, h: 64, label: '分享', kind: 'secondary', id: 'share' };
  const skinBtn: Button = { x: 140 + (DESIGN_W - 260) / 2, y: py + 500, w: (DESIGN_W - 260) / 2, h: 64, label: '皮肤', kind: 'secondary', id: 'skins' };

  drawButton(ctx, nextBtn, false);
  drawButton(ctx, shareBtn, false);
  drawButton(ctx, skinBtn, false);

  return [nextBtn, shareBtn, skinBtn];
}
```

### 14.4 重写 `src/main.ts`（完整流程）

```ts
import { platform } from './platform';
import { login, fetchConfig, reportProgress, reportSkip, reportEvent } from './net/api';
import { Session } from './game/session';
import { LevelStore, ProgressStore } from './game/progressStore';
import { DESIGN_W, DESIGN_H, BOARD_TOP } from './render/layout';
import { drawBoard } from './render/board';
import { drawTrail } from './render/trail';
import { drawHud, drawProgressBar, drawResult, hitButton, type Button } from './ui/hud';
import { findHint } from './core/solver';
import { CellType } from './core/types';

const PALETTE = {
  block: '#39415C',
  brick: '#5A6480',
  brickLit: '#6692F5',
  oneway: '#9B7BE8',
  start: '#5EA94C',
  end: '#E97E6A',
};

const { canvas, width, height, dpr } = platform.getCanvas();
const ctx = canvas.getContext('2d')!;

const scale = Math.min(width / DESIGN_W, height / DESIGN_H);
const offsetX = (width - DESIGN_W * scale) / 2;
const offsetY = (height - DESIGN_H * scale) / 2;

const session = new Session();
const levels = new LevelStore();
const progress = new ProgressStore();

let config: any = null;
let phase: 'loading' | 'playing' | 'result' = 'loading';
let resultButtons: Button[] = [];
let resultData: { levelIdx: number; steps: number; shortest: number; timeMs: number; unlockedSkin: any } | null = null;
let pressedBtn: string | null = null;
let pendingLevel = 1;

/** 画布坐标 → 设计坐标。 */
function toDesign(clientX: number, clientY: number): { x: number; y: number } {
  if (platform.name === 'wechat') {
    return { x: (clientX - offsetX) / scale, y: (clientY - offsetY) / scale };
  }
  const rect = canvas.getBoundingClientRect();
  const px = (clientX - rect.left) * (canvas.width / rect.width);
  const py = (clientY - rect.top) * (canvas.height / rect.height);
  return { x: (px / dpr - offsetX) / scale, y: (py / dpr - offsetY) / scale };
}

/** 顶部操作按钮。 */
const hintBtn: Button = { x: DESIGN_W - 210, y: 100, w: 90, h: 60, label: '提示', kind: 'ghost', id: 'hint' };
const skipBtn: Button = { x: DESIGN_W - 110, y: 100, w: 90, h: 60, label: '跳过', kind: 'ghost', id: 'skip' };

session.on((e) => {
  if (e.type === 'won') void onWin(e.steps, e.timeMs, e.path);
  if (e.type === 'break') platform.vibrate();
  if (e.type === 'moved') reportEvent('tile_light', { level_idx: session.state?.level.idx });
});

/** 通关处理：上报成绩。 */
async function onWin(steps: number, timeMs: number, path: number[]) {
  const level = session.state!.level;
  phase = 'result';
  reportEvent('level_pass', { level_idx: level.idx, steps, time_ms: timeMs, skipped: false });

  let unlockedSkin: any = null;
  try {
    const res = await reportProgress(level.idx, steps, timeMs, path);
    unlockedSkin = res.unlockedSkin;
    if (res.unlockedSkin) {
      progress.unlockSkin(res.unlockedSkin.id);
      platform.vibrate();
    }
    progress.sync({ maxLevel: res.maxLevel, unlockedSkins: res.unlockedSkins });
  } catch (e) {
    console.error('上报成绩失败', e);
  }

  resultData = {
    levelIdx: level.idx,
    steps,
    shortest: level.shortest,
    timeMs,
    unlockedSkin,
  };
  pendingLevel = level.idx + 1;
}

/** 进入关卡。 */
async function enterLevel(idx: number) {
  const lv = await levels.ensure(idx, config.levelVersion);
  if (!lv) {
    console.error('关卡加载失败', idx);
    return;
  }
  session.load(lv);
  phase = 'playing';
  reportEvent('level_start', { level_idx: idx, diff: lv.diff, w: lv.w, h: lv.h });
}

/** 使用提示。 */
async function useHint() {
  const st = session.state;
  if (!st || st.phase !== 'idle' && st.phase !== 'dragging') return;
  const head = st.path.length ? st.path[st.path.length - 1] : findStartIdx(st);
  const hint = findHint(st.level.grid, { x: head % st.level.w, y: Math.floor(head / st.level.w) }, 2);
  if (hint.length === 0) {
    console.log('当前无法给出提示');
    return;
  }
  // 广告逻辑在 T17 接入；当前直接给提示
  session.setHint(hint);
  reportEvent('hint_use', { level_idx: st.level.idx, via_ad: false });
}

/** 跳过关卡。 */
async function useSkip() {
  const st = session.state;
  if (!st) return;
  // 广告逻辑在 T17 接入
  await reportSkip(st.level.idx);
  session.markSkipped();
  await onWin(st.level.shortest, 0, []);
  reportEvent('skip_use', { level_idx: st.level.idx, via_ad: true });
}

/** 找起点索引。 */
function findStartIdx(st: any): number {
  for (let y = 0; y < st.level.h; y++) {
    for (let x = 0; x < st.level.w; x++) {
      if (st.level.grid[y][x].type === CellType.START) return y * st.level.w + x;
    }
  }
  return 0;
}

/** 指针按下。 */
function onDown(clientX: number, clientY: number) {
  const d = toDesign(clientX, clientY);
  if (phase === 'result') {
    for (const b of resultButtons) {
      if (hitButton(b, d.x, d.y)) { pressedBtn = b.id; return; }
    }
    return;
  }
  if (phase !== 'playing') return;
  if (hitButton(hintBtn, d.x, d.y)) { pressedBtn = 'hint'; return; }
  if (hitButton(skipBtn, d.x, d.y)) { pressedBtn = 'skip'; return; }
  session.pointerDown(d.x, d.y);
}

/** 指针移动。 */
function onMove(clientX: number, clientY: number) {
  if (phase !== 'playing') return;
  const d = toDesign(clientX, clientY);
  session.pointerMove(d.x, d.y);
}

/** 指针抬起。 */
function onUp() {
  if (pressedBtn) {
    const id = pressedBtn;
    pressedBtn = null;
    if (id === 'hint') void useHint();
    else if (id === 'skip') void useSkip();
    else if (id === 'next') void enterLevel(pendingLevel);
    else if (id === 'share') { void platform.share({ title: `我卡在第 ${session.state?.level.idx ?? 1} 关` }); reportEvent('share_click', {}); }
    else if (id === 'skins') { /* T16 实现皮肤页 */ }
    return;
  }
  if (phase === 'playing') session.pointerUp();
}

// Web 事件绑定
if (platform.name === 'web') {
  canvas.addEventListener('pointerdown', (e) => onDown(e.clientX, e.clientY));
  canvas.addEventListener('pointermove', (e) => onMove(e.clientX, e.clientY));
  canvas.addEventListener('pointerup', onUp);
  canvas.addEventListener('pointercancel', onUp);
}

/** 渲染循环。 */
function frame() {
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  ctx.scale(dpr * scale, dpr * scale);
  ctx.translate(offsetX / dpr, offsetY / dpr);

  ctx.fillStyle = '#141824';
  ctx.fillRect(0, 0, DESIGN_W, DESIGN_H);

  const st = session.state;
  if (st && session.layout) {
    drawBoard(ctx, st.level, session.layout, {
      lit: st.lit,
      path: st.path,
      hint: st.hint,
      shakeCell: st.shakeCell,
      shakeAt: st.shakeAt,
    }, PALETTE, Date.now());

    drawTrail(ctx, st.path, st.level.w, session.layout, '#6692F5', 'rgba(102,146,245,0.35)');

    drawHud(ctx, st.level.idx, progress.current.currentSkin, progress.current.unlockedSkins.length, config?.skins?.length ?? 13);
    drawProgressBar(ctx, progress.current.maxLevel, config?.levelCount ?? 1000);

    // 顶部按钮
    const { drawButton } = require('./ui/hud');
    drawButton(ctx, hintBtn, pressedBtn === 'hint');
    drawButton(ctx, skipBtn, pressedBtn === 'skip');
  }

  if (phase === 'result' && resultData) {
    resultButtons = drawResult(
      ctx, resultData.levelIdx, resultData.steps, resultData.shortest,
      resultData.timeMs, resultData.unlockedSkin,
      progress.current.maxLevel / (config?.levelCount ?? 1000),
    );
  }

  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

/** 启动。 */
async function boot() {
  progress.restore();
  try {
    const loginRes = await login();
    progress.sync({
      maxLevel: loginRes.maxLevel || 1,
      unlockedSkins: loginRes.unlockedSkins,
    });
    config = await fetchConfig();
    await levels.init(config.levelVersion);
    const startIdx = Math.min(progress.current.maxLevel || 1, levels.loadedMax || 1);
    await enterLevel(startIdx);
    reportEvent('app_start', {});
  } catch (e) {
    console.error('启动失败', e);
    phase = 'loading';
  }
}
```

```ts
import { describe, it, expect } from 'vitest';
import { Session } from '../src/game/session';
import { generateLevel } from '../src/core/generator';
import { solve } from '../src/core/solver';

describe('session', () => {
  it('按下非起点无效', () => {
    const s = new Session();
    const lv = generateLevel(1, 1);
    s.load(lv);
    s.pointerDown(1000, 1000);
    expect(s.state?.path.length).toBe(0);
  });

  it('沿解法路径拖动可通关', () => {
    const s = new Session();
    const lv = generateLevel(1, 1);
    s.load(lv);
    const res = solve(lv.grid);
    const path = res.paths[0];

    // 用格子中心坐标模拟拖动
    const cell = sessionCell(s);
    let won = false;
    s.on((e) => { if (e.type === 'won') won = true; });

    s.pointerDown(centerX(path[0], lv.w, cell), centerY(path[0], lv.w, cell));
    for (let i = 1; i < path.length; i++) {
      s.pointerMove(centerX(path[i], lv.w, cell), centerY(path[i], lv.w, cell));
    }
    s.pointerUp();
    expect(won).toBe(true);
    expect(s.state?.phase).toBe('won');
  });

  it('松手未到终点则路径清空', () => {
    const s = new Session();
    const lv = generateLevel(1, 1);
    s.load(lv);
    const res = solve(lv.grid);
    const path = res.paths[0];
    const cell = sessionCell(s);
    s.pointerDown(centerX(path[0], lv.w, cell), centerY(path[0], lv.w, cell));
    s.pointerMove(centerX(path[1], lv.w, cell), centerY(path[1], lv.w, cell));
    s.pointerUp();
    expect(s.state?.path.length).toBe(0);
    expect(s.state?.phase).toBe('idle');
  });
});

function sessionCell(s: Session): number {
  return s.layout!.cell;
}
function centerX(idx: number, w: number, cell: number): number {
  return s_originX(s) + (idx % w) * cell + cell / 2;
}
function centerY(idx: number, w: number, cell: number): number {
  return s_originY(s) + Math.floor(idx / w) * cell + cell / 2;
}
function s_originX(s: Session): number { return s.layout!.originX; }
function s_originY(s: Session): number { return s.layout!.originY; }
```
