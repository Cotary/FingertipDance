### 20.1 写 `src/game/challenge.ts`

```ts
import { platform } from '../platform';
import { reportChallenge } from '../net/api';
import type { Level } from '../core/types';

/** 挑战上下文。 */
export interface Challenge {
  /** 被挑战的关号 */
  levelIdx: number;
  /** 挑战者的 uid */
  fromUid: number;
  /** 挑战者的昵称 */
  fromName: string;
  /** 挑战者的步数（若已知） */
  fromSteps: number;
}

/**
 * 从启动参数解析挑战信息。
 * 微信：wx.getLaunchOptionsSync().query
 * Web：URL 的 search 参数
 */
export function parseChallenge(): Challenge | null {
  let query: Record<string, string> = {};

  if (platform.name === 'wechat') {
    const wx = (globalThis as any).wx;
    const opts = wx?.getLaunchOptionsSync?.();
    query = opts?.query ?? {};
  } else if (typeof location !== 'undefined') {
    query = Object.fromEntries(new URLSearchParams(location.search).entries());
  }

  const lvl = query.l;
  if (!lvl) return null;
  const levelIdx = parseInt(lvl, 10);
  if (!Number.isFinite(levelIdx) || levelIdx < 1) return null;

  return {
    levelIdx,
    fromUid: parseInt(query.uid ?? '0', 10) || 0,
    fromName: query.name ? decodeURIComponent(query.name) : '好友',
    fromSteps: parseInt(query.steps ?? '0', 10) || 0,
  };
}

/**
 * 构造分享链接。
 */
export function buildShareQuery(challenge: { levelIdx: number; steps?: number }): string {
  const parts = [`l=${challenge.levelIdx}`];
  if (challenge.steps) parts.push(`steps=${challenge.steps}`);
  return parts.join('&');
}
```

### 20.3 写 `src/ui/share.ts`（分享海报）

```ts
import { DESIGN_W, DESIGN_H } from '../render/layout';
import { roundRect } from '../render/board';
import { COLORS } from './hud';
import { paintSkin, skinById } from '../render/skins';
import type { Level } from '../core/types';

/** 分享文案池。 */
const COPY_POOL = [
  '我卡在第 {lv} 关，你能连过去吗？',
  '这关我用了 {steps} 步，你呢？',
  '敢不敢试试我的皮肤？',
  '第 {lv} 关，我不信你过得去',
];

/**
 * 生成分享文案。
 */
export function shareCopy(levelIdx: number, steps: number): string {
  const tpl = COPY_POOL[Math.floor(Math.random() * COPY_POOL.length)];
  return tpl.replace('{lv}', String(levelIdx)).replace('{steps}', String(steps));
}

/**
 * 绘制分享海报到离屏 canvas。
 */
export function renderPoster(level: Level, skinId: string, steps: number, nickname: string): HTMLCanvasElement {
  const canvas = document.createElement('canvas');
  canvas.width = 750;
  canvas.height = 1334;
  const ctx = canvas.getContext('2d')!;

  // 背景
  const bg = ctx.createLinearGradient(0, 0, 750, 1334);
  bg.addColorStop(0, '#141824');
  bg.addColorStop(1, '#1E2436');
  ctx.fillStyle = bg;
  ctx.fillRect(0, 0, 750, 1334);

  const skin = skinById(skinId);

  // 顶部标题
  ctx.fillStyle = COLORS.text;
  ctx.font = '700 64px system-ui, sans-serif';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'top';
  ctx.fillText(`第 ${level.idx} 关`, 375, 120);

  // 皮肤预览（用真实棋盘片段）
  const pvSize = 420;
  const pvX = (750 - pvSize) / 2;
  const pvY = 240;
  const cell = pvSize / 5;
  for (let gy = 0; gy < 5; gy++) {
    for (let gx = 0; gx < 5; gx++) {
      const idx = gy * 5 + gx;
      const size = cell - 6;
      paintSkin(ctx, skin, pvX + gx * cell + 3, pvY + gy * cell + 3, size, size, idx + 1, 0);
    }
  }

  // 文案
  ctx.fillStyle = COLORS.yellow;
  ctx.font = '600 38px system-ui, sans-serif';
  ctx.fillText(shareCopy(level.idx, steps), 375, 740);

  // 步数
  ctx.fillStyle = COLORS.text;
  ctx.font = '30px system-ui, sans-serif';
  ctx.fillText(`我的步数 ${steps}`, 375, 810);

  // 昵称
  ctx.fillStyle = COLORS.subText;
  ctx.font = '26px system-ui, sans-serif';
  ctx.fillText(nickname || '我', 375, 870);

  // 底部 CTA
  ctx.fillStyle = COLORS.blue;
  roundRect(ctx, 175, 1080, 400, 100, 50);
  ctx.fill();
  ctx.fillStyle = '#FFFFFF';
  ctx.font = '600 34px system-ui, sans-serif';
  ctx.textBaseline = 'middle';
  ctx.fillText('来试试看', 375, 1130);

  return canvas;
}
```

```ts
async function onChallengeWin(steps: number, timeMs: number, path: number[]) {
  const level = session.state!.level;
  let isNewBest = false;
  let best = { steps: level.shortest, timeMs: 0 };
  try {
    const res = await reportChallenge(level.idx, steps, timeMs, path);
    isNewBest = res.isNewBest;
    best = res.best;
  } catch (e) {
    console.error('提交挑战成绩失败', e);
  }

  reportEvent('challenge_result', {
    from_uid: challenge?.fromUid, level_idx: level.idx, win: steps <= best.steps,
  });

  winAnimStart = Date.now();
  fx.wave(path);
  setTimeout(() => {
    phase = 'result';
    challengeResult = { win: steps <= best.steps, mySteps: steps, bestSteps: best.steps, fromName: challenge?.fromName ?? '好友' };
  }, 1200);
}
```

```ts
function doShare() {
  const st = session.state;
  if (!st) return;
  const query = buildShareQuery({ levelIdx: st.level.idx, steps: st.level.shortest });
  const poster = renderPoster(st.level, progress.current.currentSkin, st.level.shortest, '我');
  // Web：转 base64 或直接下载
  if (platform.name === 'web') {
    const url = poster.toDataURL('image/png');
    const a = document.createElement('a');
    a.href = url;
    a.download = `level-${st.level.idx}.png`;
    a.click();
  }
  void platform.share({ title: shareCopy(st.level.idx, st.level.shortest), query });
  reportEvent('share_click', { level_idx: st.level.idx });
}
```

## 坑
- **挑战模式不能推进主线进度** → 走 `/api/challenge` 而不是 `/api/progress`
- **`parseChallenge` 在 Web 端读 `location.search`** → 微信端读 `wx.getLaunchOptionsSync().query`
- **海报 canvas 在微信端** → `document.createElement('canvas')` 不存在 → 需要用 `wx.createOffscreenCanvas({type:'2d', width, height})`（基础库 2.16.1+）
  → **在 `renderPoster` 里做平台分支**：
  ```ts
  const canvas = platform.name === 'wechat'
    ? wx.createOffscreenCanvas({ type: '2d', width: 750, height: 1334 })
    : document.createElement('canvas');
  ```
- **海报导出** → Web 用 `toDataURL`；微信用 `wx.canvasToTempFilePath`（需要 `canvas` 字段指定）
- **`Math.random()` 用于文案池** → UI 层允许（`core/` 层才禁止）
