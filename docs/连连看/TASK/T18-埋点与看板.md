### 18.1 写 `src/analytics/tracker.ts`

```ts
import { platform } from '../platform';
import { reportEvent } from '../net/api';

/** 埋点事件名。 */
export type EventName =
  | 'app_start' | 'login_success' | 'level_start' | 'tile_light' | 'level_pass'
  | 'level_fail' | 'level_abandon' | 'hint_use' | 'skip_use'
  | 'ad_start' | 'ad_complete' | 'ad_abort' | 'ad_error'
  | 'skin_unlock' | 'skin_equip' | 'skin_page_view'
  | 'leaderboard_view' | 'share_click' | 'share_success'
  | 'challenge_start' | 'challenge_result';

/** 缓冲的事件。 */
interface BufferedEvent {
  name: EventName;
  ts: number;
  params: Record<string, unknown>;
}

const KEY_BUFFER = 'event_buffer';

/**
 * 埋点追踪器。本地缓冲 + 批量上报 + 断网补传。
 */
export class Tracker {
  private buffer: BufferedEvent[] = [];

  /** 初始化：从本地恢复缓冲。 */
  init(): void {
    const raw = platform.storage.get(KEY_BUFFER);
    if (!raw) return;
    try {
      const arr = JSON.parse(raw) as BufferedEvent[];
      if (Array.isArray(arr)) this.buffer = arr.slice(0, 200);
    } catch { /* 忽略 */ }
    // 恢复后立即补传
    if (this.buffer.length > 0) void this.flush();
  }

  /**
   * 记录一个事件。
   */
  track(name: EventName, params: Record<string, unknown> = {}): void {
    this.buffer.push({ name, ts: Date.now(), params });
    if (this.buffer.length >= 20) {
      void this.flush();
    } else {
      this.persist();
    }
  }

  /**
   * 批量上报。
   */
  async flush(): Promise<void> {
    if (this.buffer.length === 0) return;
    const batch = this.buffer.slice(0, 50);
    try {
      await platform.request('/api/report', 'POST', { events: batch });
      this.buffer.splice(0, batch.length);
      this.persist();
      if (this.buffer.length > 0) setTimeout(() => void this.flush(), 1000);
    } catch {
      // 上报失败 → 保留在缓冲里，下次重试
      this.persist();
    }
  }

  /** 持久化缓冲。 */
  private persist(): void {
    try {
      platform.storage.set(KEY_BUFFER, JSON.stringify(this.buffer.slice(-200)));
    } catch { /* 存储满了就放弃 */ }
  }
}

/** 全局单例。 */
export const tracker = new Tracker();

/**
 * 便捷函数。保留与 `net/api.reportEvent` 的兼容。
 */
export function track(name: EventName, params: Record<string, unknown> = {}): void {
  tracker.track(name, params);
}
```

### 18.5 写后端分析脚本 `server/tools/analyze/main.go`

```go
package main

import (
	"database/sql"
	"encoding/json"
	"fmt"
	"os"
	"sort"
	"time"

	_ "github.com/go-sql-driver/mysql"
)

func main() {
	dsn := os.Getenv("DB_DSN")
	if dsn == "" {
		dsn = "root:root@tcp(127.0.0.1:3306)/fgame?charset=utf8mb4&parseTime=true"
	}
	db, err := sql.Open("mysql", dsn)
	if err != nil {
		fmt.Println("连接失败:", err)
		os.Exit(1)
	}
	defer db.Close()

	fmt.Println("========== 关卡漏斗 ==========")
	printLevelFunnel(db)

	fmt.Println("\n========== 广告漏斗 ==========")
	printAdFunnel(db)

	fmt.Println("\n========== 分享漏斗 ==========")
	printShareFunnel(db)

	fmt.Println("\n========== 皮肤解锁 ==========")
	printSkinFunnel(db)
}

// printLevelFunnel 打印每关的开始/通过/放弃率。
func printLevelFunnel(db *sql.DB) {
	rows, err := db.Query(`
		SELECT
		  JSON_UNQUOTE(JSON_EXTRACT(params, '$.level_idx')) AS lvl,
		  SUM(CASE WHEN name='level_start' THEN 1 ELSE 0 END) AS starts,
		  SUM(CASE WHEN name='level_pass' THEN 1 ELSE 0 END) AS passes,
		  SUM(CASE WHEN name='level_fail' THEN 1 ELSE 0 END) AS fails
		FROM events
		WHERE name IN ('level_start','level_pass','level_fail')
		GROUP BY lvl ORDER BY lvl
	`)
	if err != nil {
		fmt.Println("查询失败:", err)
		return
	}
	defer rows.Close()

	fmt.Printf("%-8s %-8s %-8s %-10s %-10s\n", "关卡", "开始", "通过", "通过率", "失败率")
	for rows.Next() {
		var lvl sql.NullInt64
		var starts, passes, fails sql.NullInt64
		if err := rows.Scan(&lvl, &starts, &passes, &fails); err != nil {
			continue
		}
		passRate := 0.0
		if starts.Int64 > 0 {
			passRate = float64(passes.Int64) / float64(starts.Int64) * 100
		}
		failRate := 0.0
		if starts.Int64 > 0 {
			failRate = float64(fails.Int64) / float64(starts.Int64) * 100
		}
		fmt.Printf("%-8d %-8d %-8d %-9.1f%% %-9.1f%%\n",
			lvl.Int64, starts.Int64, passes.Int64, passRate, failRate)
	}
}

// printAdFunnel 打印广告完播率。
func printAdFunnel(db *sql.DB) {
	rows, err := db.Query(`
		SELECT
		  JSON_UNQUOTE(JSON_EXTRACT(params, '$.placement')) AS placement,
		  SUM(CASE WHEN name='ad_start' THEN 1 ELSE 0 END) AS starts,
		  SUM(CASE WHEN name='ad_complete' THEN 1 ELSE 0 END) AS completes,
		  SUM(CASE WHEN name='ad_abort' THEN 1 ELSE 0 END) AS aborts
		FROM events
		WHERE name LIKE 'ad_%'
		GROUP BY placement
	`)
	if err != nil {
		fmt.Println("查询失败:", err)
		return
	}
	defer rows.Close()

	fmt.Printf("%-22s %-8s %-8s %-8s %-10s\n", "广告位", "开始", "完成", "中断", "完播率")
	for rows.Next() {
		var placement sql.NullString
		var starts, completes, aborts sql.NullInt64
		if err := rows.Scan(&placement, &starts, &completes, &aborts); err != nil {
			continue
		}
		rate := 0.0
		if starts.Int64 > 0 {
			rate = float64(completes.Int64) / float64(starts.Int64) * 100
		}
		fmt.Printf("%-22s %-8d %-8d %-8d %-9.1f%%\n",
			placement.String, starts.Int64, completes.Int64, aborts.Int64, rate)
	}
}

// printShareFunnel 打印分享漏斗。
func printShareFunnel(db *sql.DB) {
	rows, err := db.Query(`
		SELECT name, COUNT(*) FROM events
		WHERE name IN ('share_click','share_success','challenge_start','challenge_result')
		GROUP BY name
	`)
	if err != nil {
		fmt.Println("查询失败:", err)
		return
	}
	defer rows.Close()
	for rows.Next() {
		var name string
		var cnt int64
		if err := rows.Scan(&name, &cnt); err != nil {
			continue
		}
		fmt.Printf("%-20s %d\n", name, cnt)
	}
}

// printSkinFunnel 打印皮肤解锁分布。
func printSkinFunnel(db *sql.DB) {
	rows, err := db.Query(`
		SELECT JSON_UNQUOTE(JSON_EXTRACT(params, '$.via')) AS via, COUNT(*) 
		FROM events WHERE name='skin_unlock' GROUP BY via
	`)
	if err != nil {
		fmt.Println("查询失败:", err)
		return
	}
	defer rows.Close()
	for rows.Next() {
		var via sql.NullString
		var cnt int64
		if err := rows.Scan(&via, &cnt); err != nil {
			continue
		}
		fmt.Printf("解锁方式 %-12s %d 次\n", via.String, cnt)
	}

	// 当前持有皮肤分布
	fmt.Println("\n当前玩家皮肤持有分布：")
	rows2, err := db.Query(`
		SELECT JSON_LENGTH(unlocked_skins) AS cnt, COUNT(*) AS players
		FROM users GROUP BY cnt ORDER BY cnt
	`)
	if err != nil {
		return
	}
	defer rows2.Close()
	for rows2.Next() {
		var cnt, players int64
		if err := rows2.Scan(&cnt, &players); err != nil {
			continue
		}
		fmt.Printf("  %2d 个皮肤: %d 人\n", cnt, players)
	}
}

// unused 保留，避免编译告警。
var _ = json.Marshal
var _ = sort.Ints
var _ = time.Now
```

## 坑
- **`reportEvent` 签名改造** → `net/api.ts` 与 `analytics/tracker.ts` 循环依赖 → `tracker` 只 import `platform`，不 import `net/api`；`net/api` import `tracker` → 单向依赖，安全
- **`events` 表增长快** → 一个月百万级没问题，但要定期归档（加 `ts < 某日期` 的清理任务）
- **`JSON_EXTRACT` 在 MySQL 5.7+ 可用** → MySQL 8 没问题
- **缓冲的埋点在退出时可能丢** → 微信端切后台会销毁 → 在 `onHide` 时调 `tracker.flush()`
- **`platform.request` 失败不抛**（微信端 reject 了）→ `tracker.flush` 的 catch 已处理
