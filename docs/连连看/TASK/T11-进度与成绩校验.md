### 11.1 写 `internal/games/line/validate.go`（路径合法性校验）

**规则必须与 TypeScript 侧 `isPathLegal` 完全一致**（见 `02-关卡格式与算法规格.md` 第 5.4 节）。

```go
package line

import (
	"fmt"
)

// 方向增量，与 TypeScript 侧 DX/DY 一致。
var dirDX = [4]int{0, 0, -1, 1}
var dirDY = [4]int{-1, 1, 0, 0}

// PathValidationError 描述路径校验失败的原因。
type PathValidationError struct {
	Reason string
}

// Error 实现 error 接口。
func (e *PathValidationError) Error() string { return e.Reason }

// ValidatePath 校验一条路径是否合法。
// @param cells 解码后的格子数组
// @param w 棋盘列数
// @param path 格子索引序列
func ValidatePath(cells []Cell, w int, path []int) error {
	if len(path) < 2 {
		return &PathValidationError{Reason: "路径长度不足"}
	}
	if len(path) > 200 {
		return &PathValidationError{Reason: "路径过长"}
	}

	// 找到起点与终点
	startIdx, endIdx := -1, -1
	for i, c := range cells {
		switch c.Type {
		case CellStart:
			startIdx = i
		case CellEnd:
			endIdx = i
		}
	}
	if startIdx < 0 || endIdx < 0 {
		return &PathValidationError{Reason: "关卡缺少起点或终点"}
	}
	if path[0] != startIdx {
		return &PathValidationError{Reason: "路径起点不匹配"}
	}
	if path[len(path)-1] != endIdx {
		return &PathValidationError{Reason: "路径终点不匹配"}
	}

	for i, idx := range path {
		if idx < 0 || idx >= len(cells) {
			return &PathValidationError{Reason: fmt.Sprintf("第 %d 步索引越界", i)}
		}
		cell := cells[idx]

		// 起点终点不能出现在中间
		if i > 0 && i < len(path)-1 {
			if cell.Type == CellStart || cell.Type == CellEnd {
				return &PathValidationError{Reason: fmt.Sprintf("第 %d 步经过起点或终点", i)}
			}
		}
		if cell.Type == CellBlock {
			return &PathValidationError{Reason: fmt.Sprintf("第 %d 步穿过阻挡块", i)}
		}

		if i > 0 {
			prev := path[i-1]
			dx := (idx % w) - (prev % w)
			dy := (idx / w) - (prev / w)
			if abs(dx)+abs(dy) != 1 {
				return &PathValidationError{Reason: fmt.Sprintf("第 %d 步与上一步不相邻", i)}
			}
			enterDir := dirOfDelta(dx, dy)
			if cell.Type == CellOneway && int(cell.Dir) != enterDir {
				return &PathValidationError{Reason: fmt.Sprintf("第 %d 步违背单向块方向", i)}
			}
		}
	}
	return nil
}

// dirOfDelta 由坐标差求方向。
func dirOfDelta(dx, dy int) int {
	switch {
	case dy == -1:
		return DirUp
	case dy == 1:
		return DirDown
	case dx == -1:
		return DirLeft
	default:
		return DirRight
	}
}

// abs 返回绝对值。
func abs(v int) int {
	if v < 0 {
		return -v
	}
	return v
}
```

### 11.2 写 `internal/games/line/progress.go`

```go
package line

import (
	"database/sql"
	"encoding/json"
	"fmt"
	"time"

	"github.com/gin-gonic/gin"

	"github.com/tupss/f-game/server/internal/core"
)

// 进度上报请求。
type progressReq struct {
	LevelIdx int   `json:"levelIdx"`
	Steps    int   `json:"steps"`
	TimeMs   int   `json:"timeMs"`
	Path     []int `json:"path"`
	Skipped  bool  `json:"skipped"`
}

// 解锁的皮肤信息。
type unlockedSkinInfo struct {
	ID   string `json:"id"`
	Name string `json:"name"`
}

// ProgressHandler 上报通关成绩，校验合法性并推进进度。
func ProgressHandler(db *core.DB, skins []SkinConfig) gin.HandlerFunc {
	skinMap := make(map[string]string, len(skins))
	for _, s := range skins {
		skinMap[s.ID] = s.Name
	}

	return func(c *gin.Context) {
		var req progressReq
		if err := c.ShouldBindJSON(&req); err != nil {
			core.Fail(c, core.CodeBadRequest, "参数错误: "+err.Error())
			return
		}
		uid := core.UIDFromContext(c)

		lv := Levels.Get(req.LevelIdx)
		if lv == nil {
			core.Fail(c, core.CodeLevelNotFound, "关卡不存在")
			return
		}

		// ── 校验 ──
		if req.Skipped {
			// 跳过：步数必须等于最短步数
			if req.Steps != lv.Shortest {
				core.Fail(c, core.CodeStepsTooLow, "跳过时步数必须等于最短步数")
				return
			}
		} else {
			cells, err := decodeGridBase64(lv.Grid)
			if err != nil {
				core.Fail(c, core.CodeInternal, "解码关卡失败: "+err.Error())
				return
			}
			if len(req.Path) != req.Steps+1 {
				core.Fail(c, core.CodePathIllegal, "路径长度与步数不匹配")
				return
			}
			if err := ValidatePath(cells, lv.W, req.Path); err != nil {
				core.Fail(c, core.CodePathIllegal, "路径非法: "+err.Error())
				return
			}
			if req.Steps < lv.Shortest {
				core.Fail(c, core.CodeStepsTooLow, fmt.Sprintf("步数 %d 低于理论最短 %d", req.Steps, lv.Shortest))
				return
			}
			if req.TimeMs < req.Steps*50 {
				core.Fail(c, core.CodeTimeInvalid, "用时异常")
				return
			}
		}

		// ── 事务写入 ──
		res, err := applyProgress(db, uid, lv, req)
		if err != nil {
			core.Fail(c, core.CodeInternal, "写入进度失败: "+err.Error())
			return
		}

		core.OK(c, res)
	}
}

// 进度写入结果。
type progressResult struct {
	Passed        bool                `json:"passed"`
	UnlockedSkin  *unlockedSkinInfo   `json:"unlockedSkin"`
	IsNewBest     bool                `json:"isNewBest"`
	MaxLevel      int                 `json:"maxLevel"`
	UnlockedSkins []string            `json:"unlockedSkins"`
	MyRank        int64               `json:"myRank"`
}

// applyProgress 在事务内更新关卡记录、用户进度、排行榜与皮肤。
func applyProgress(db *core.DB, uid int64, lv *SerializedLevel, req progressReq) (*progressResult, error) {
	tx, err := db.Begin()
	if err != nil {
		return nil, fmt.Errorf("开启事务: %w", err)
	}
	defer tx.Rollback()

	now := time.Now()

	// 查已有记录，判断是否首次通关与是否新纪录
	var oldSteps, oldTime int
	var oldSkipped bool
	err = tx.QueryRow(
		"SELECT steps, time_ms, skipped FROM level_records WHERE uid=? AND level_idx=?",
		uid, lv.Idx,
	).Scan(&oldSteps, &oldTime, &oldSkipped)
	firstPass := errors.Is(err, sql.ErrNoRows)
	if err != nil && !firstPass {
		return nil, fmt.Errorf("查询关卡记录: %w", err)
	}

	isNewBest := firstPass || req.Steps < oldSteps || (req.Steps == oldSteps && req.TimeMs < oldTime)

	pathJSON, _ := json.Marshal(req.Path)
	skipped := 0
	if req.Skipped {
		skipped = 1
	}

	if firstPass {
		_, err = tx.Exec(
			"INSERT INTO level_records (uid, level_idx, steps, time_ms, path, skipped, created_at) VALUES (?,?,?,?,?,?,?)",
			uid, lv.Idx, req.Steps, req.TimeMs, string(pathJSON), skipped, now,
		)
	} else if isNewBest {
		_, err = tx.Exec(
			"UPDATE level_records SET steps=?, time_ms=?, path=?, skipped=? WHERE uid=? AND level_idx=?",
			req.Steps, req.TimeMs, string(pathJSON), skipped, uid, lv.Idx,
		)
	}
	if err != nil {
		return nil, fmt.Errorf("写入关卡记录: %w", err)
	}

	// 累计总步数与总用时（仅首次通关累加，避免重复提交刷数值）
	if firstPass {
		_, err = tx.Exec(
			"UPDATE users SET total_steps = total_steps + ?, total_time_ms = total_time_ms + ? WHERE id=?",
			req.Steps, req.TimeMs, uid,
		)
		if err != nil {
			return nil, fmt.Errorf("累计总步数: %w", err)
		}
	}

	// 推进最高关卡（跳过的关卡也算推进，否则玩家会卡死）
	if lv.Idx > 0 {
		_, err = tx.Exec(
			"UPDATE users SET max_level = GREATEST(max_level, ?) WHERE id=?",
			lv.Idx, uid,
		)
		if err != nil {
			return nil, fmt.Errorf("推进最高关卡: %w", err)
		}
	}

	// 皮肤解锁：仅在真正通关（未跳过）且该关有关联皮肤时发放
	var unlocked *unlockedSkinInfo
	if !req.Skipped && lv.UnlockSkin != nil && *lv.UnlockSkin != "" {
		var unlockedJSON string
		row := tx.QueryRow("SELECT unlocked_skins FROM users WHERE id=?", uid)
		if err := row.Scan(&unlockedJSON); err != nil {
			return nil, fmt.Errorf("查询皮肤: %w", err)
		}
		var list []string
		_ = json.Unmarshal([]byte(unlockedJSON), &list)
		if list == nil {
			list = []string{}
		}
		found := false
		for _, id := range list {
			if id == *lv.UnlockSkin {
				found = true
				break
			}
		}
		if !found {
			list = append(list, *lv.UnlockSkin)
			newJSON, _ := json.Marshal(list)
			_, err = tx.Exec("UPDATE users SET unlocked_skins=? WHERE id=?", string(newJSON), uid)
			if err != nil {
				return nil, fmt.Errorf("写入皮肤: %w", err)
			}
			unlocked = &unlockedSkinInfo{ID: *lv.UnlockSkin, Name: skinMap[*lv.UnlockSkin]}
		}
	}

	// 成绩流水
	_, err = tx.Exec(
		"INSERT INTO scores (uid, level_idx, steps, time_ms, created_at) VALUES (?,?,?,?,?)",
		uid, lv.Idx, req.Steps, req.TimeMs, now,
	)
	if err != nil {
		return nil, fmt.Errorf("写入成绩流水: %w", err)
	}

	if err := tx.Commit(); err != nil {
		return nil, fmt.Errorf("提交事务: %w", err)
	}

	// 读取最新进度用于响应
	var maxLevel int
	var unlockedJSON string
	if err := db.QueryRow("SELECT max_level, unlocked_skins FROM users WHERE id=?", uid).Scan(&maxLevel, &unlockedJSON); err != nil {
		return nil, fmt.Errorf("读取进度: %w", err)
	}
	var skinList []string
	_ = json.Unmarshal([]byte(unlockedJSON), &skinList)

	return &progressResult{
		Passed:        true,
		UnlockedSkin:  unlocked,
		IsNewBest:     isNewBest,
		MaxLevel:      maxLevel,
		UnlockedSkins: skinList,
	}, nil
}
```

```go
package line

import (
	"fmt"

	"github.com/gin-gonic/gin"

	"github.com/tupss/f-game/server/internal/core"
)

// ChallengeHandler 提交挑战成绩。不推进主线进度。
func ChallengeHandler(db *core.DB) gin.HandlerFunc {
	return func(c *gin.Context) {
		var req progressReq
		if err := c.ShouldBindJSON(&req); err != nil {
			core.Fail(c, core.CodeBadRequest, "参数错误: "+err.Error())
			return
		}
		uid := core.UIDFromContext(c)

		lv := Levels.Get(req.LevelIdx)
		if lv == nil {
			core.Fail(c, core.CodeLevelNotFound, "关卡不存在")
			return
		}

		cells, err := decodeGridBase64(lv.Grid)
		if err != nil {
			core.Fail(c, core.CodeInternal, "解码关卡失败: "+err.Error())
			return
		}
		if len(req.Path) != req.Steps+1 {
			core.Fail(c, core.CodePathIllegal, "路径长度与步数不匹配")
			return
		}
		if err := ValidatePath(cells, lv.W, req.Path); err != nil {
			core.Fail(c, core.CodePathIllegal, "路径非法: "+err.Error())
			return
		}
		if req.Steps < lv.Shortest {
			core.Fail(c, core.CodeStepsTooLow, "步数低于理论最短")
			return
		}
		if req.TimeMs < req.Steps*50 {
			core.Fail(c, core.CodeTimeInvalid, "用时异常")
			return
		}

		// 只写流水，不动进度
		_, err = db.Exec(
			"INSERT INTO scores (uid, level_idx, steps, time_ms, created_at) VALUES (?,?,?,?,?)",
			uid, lv.Idx, req.Steps, req.TimeMs, nowFunc(),
		)
		if err != nil {
			core.Fail(c, core.CodeInternal, "写入成绩失败: "+err.Error())
			return
		}

		// 查询该关的历史最佳
		var bestSteps, bestTime int
		err = db.QueryRow(
			"SELECT steps, time_ms FROM scores WHERE level_idx=? ORDER BY steps ASC, time_ms ASC LIMIT 1",
			req.LevelIdx,
		).Scan(&bestSteps, &bestTime)
		if err != nil {
			core.Fail(c, core.CodeInternal, "查询最佳成绩失败: "+err.Error())
			return
		}

		core.OK(c, gin.H{
			"passed": true,
			"best": gin.H{
				"steps":  bestSteps,
				"timeMs": bestTime,
			},
			"isNewBest": req.Steps < bestSteps || (req.Steps == bestSteps && req.TimeMs < bestTime),
		})
	}
}
```

### 11.4 写 `internal/games/line/skip.go`

```go
package line

import (
	"fmt"
	"time"

	"github.com/gin-gonic/gin"

	"github.com/tupss/f-game/server/internal/core"
)

// SkipHandler 处理看广告跳过关卡。
func SkipHandler(db *core.DB) gin.HandlerFunc {
	return func(c *gin.Context) {
		var req struct {
			LevelIdx int `json:"levelIdx"`
		}
		if err := c.ShouldBindJSON(&req); err != nil {
			core.Fail(c, core.CodeBadRequest, "参数错误: "+err.Error())
			return
		}
		uid := core.UIDFromContext(c)

		lv := Levels.Get(req.LevelIdx)
		if lv == nil {
			core.Fail(c, core.CodeLevelNotFound, "关卡不存在")
			return
		}

		day := time.Now().Format("2006-01-02")
		// 幂等插入：已存在则跳过
		res, err := db.Exec(
			"INSERT IGNORE INTO skips (uid, level_idx, day) VALUES (?,?,?)",
			uid, lv.Idx, day,
		)
		if err != nil {
			core.Fail(c, core.CodeInternal, "写入跳过记录失败: "+err.Error())
			return
		}
		if n, _ := res.RowsAffected(); n == 0 {
			core.Fail(c, core.CodeSkipLimit, "本关今日已跳过")
			return
		}

		// 写关卡记录（标记为跳过）
		pathJSON := "[]"
		_, err = db.Exec(
			"INSERT INTO level_records (uid, level_idx, steps, time_ms, path, skipped, created_at) VALUES (?,?,?,?,?,1,?) "+
				"ON DUPLICATE KEY UPDATE skipped=1",
			uid, lv.Idx, lv.Shortest, 0, pathJSON, time.Now(),
		)
		if err != nil {
			core.Fail(c, core.CodeInternal, "写入关卡记录失败: "+err.Error())
			return
		}

		// 推进最高关卡（否则玩家卡死），但不发皮肤
		_, err = db.Exec("UPDATE users SET max_level = GREATEST(max_level, ?) WHERE id=?", lv.Idx, uid)
		if err != nil {
			core.Fail(c, core.CodeInternal, "推进进度失败: "+err.Error())
			return
		}

		var maxLevel int
		_ = db.QueryRow("SELECT max_level FROM users WHERE id=?", uid).Scan(&maxLevel)

		core.OK(c, gin.H{
			"ok":       true,
			"maxLevel": maxLevel,
			"skipped":  true,
		})
	}
}
```

```go
package line

import "testing"

func TestValidatePath(t *testing.T) {
	// 2x2: START(0) BRICK(1) / END(2) BRICK(3)
	cells := []Cell{
		{Type: CellStart},
		{Type: CellBrick},
		{Type: CellEnd},
		{Type: CellBrick},
	}
	if err := ValidatePath(cells, 2, []int{0, 1, 3, 2}); err != nil {
		t.Fatalf("合法路径应通过: %v", err)
	}
	if err := ValidatePath(cells, 2, []int{0, 3, 2}); err == nil {
		t.Fatal("非相邻路径应被拒绝")
	}
	if err := ValidatePath(cells, 2, []int{1, 3, 2}); err == nil {
		t.Fatal("起点不匹配应被拒绝")
	}
}

func TestValidatePathBlock(t *testing.T) {
	cells := []Cell{
		{Type: CellStart},
		{Type: CellBlock},
		{Type: CellEnd},
		{Type: CellBrick},
	}
	if err := ValidatePath(cells, 2, []int{0, 1, 2}); err == nil {
		t.Fatal("穿过阻挡块应被拒绝")
	}
}

func TestValidatePathOneway(t *testing.T) {
	// 格子 1 是单向块，箭头朝左（只能向左进入）
	cells := []Cell{
		{Type: CellStart},
		{Type: CellOneway, Dir: DirLeft},
		{Type: CellEnd},
		{Type: CellBrick},
	}
	// 从 0 进入 1 的方向是 LEFT（向右走）→ 箭头朝左意味着只能向左进入 → 不合法

	cells[1].Dir = DirRight
	if err := ValidatePath(cells, 2, []int{0, 1, 3, 2}); err != nil {
		t.Fatalf("方向正确的单向块应通过: %v", err)
	}
}
```

## 坑
- **`errors.Is(err, sql.ErrNoRows)`** 需要 `import "errors"`
- **`GREATEST(max_level, ?)`** 是 MySQL 函数，SQLite 不支持 → 本项目用 MySQL，没问题
- **`ON DUPLICATE KEY UPDATE`** 需要主键 `(uid, level_idx)` → 已在迁移里定义
- 跳过的关卡**推进 max_level 但不发皮肤** → `applyProgress` 里 `if !req.Skipped && lv.UnlockSkin != nil` 这个条件不能省
- `level_records` 的 `path` 是 JSON 列 → 插入时要 `string(pathJSON)`（MySQL 会隐式转换，但显式更安全）
- **客户端与服务端的 `isPathLegal` 必须完全一致** → 任何一侧改了规则，另一侧也要改。建议在 `docs/连连看/03-后端规格.md` 里标注「与 `core/solver.ts` 同步」
