### 9.1 写 `internal/games/line/levels.go`

```go
package line

import (
	_ "embed"
	"encoding/json"
	"fmt"
	"os"
	"sort"
	"sync"
)

// 关卡表文件（公开，会下发给客户端）。
type LevelTableFile struct {
	Version     string          `json:"version"`
	Count       int             `json:"count"`
	GeneratedAt string          `json:"generatedAt"`
	Levels      []SerializedLevel `json:"levels"`
}

// 序列化关卡。grid 是 base64 编码的紧凑网格。
type SerializedLevel struct {
	Idx        int     `json:"idx"`
	Seed       uint32  `json:"seed"`
	W          int     `json:"w"`
	H          int     `json:"h"`
	Grid       string  `json:"grid"`
	Shortest   int     `json:"shortest"`
	Turns      int     `json:"turns"`
	Blocks     int     `json:"blocks"`
	Oneways    int     `json:"oneways"`
	Diff       int     `json:"diff"`
	UnlockSkin *string `json:"unlockSkin"`
}

// 私密解法表（绝不下发）。
type SolutionTableFile struct {
	Version string `json:"version"`
	Levels  []struct {
		Idx           int    `json:"idx"`
		Solution      []int  `json:"solution"`
		SolutionSteps int    `json:"solutionSteps"`
		Turns         int    `json:"turns"`
		Diff          int    `json:"diff"`
	} `json:"levels"`
}

//go:embed levels.json
var embeddedLevels []byte

//go:embed solutions.json
var embeddedSolutions []byte

// LevelTable 提供关卡的只读访问。
type LevelTable struct {
	table     *LevelTableFile
	solutions map[int][]int
	byIdx     map[int]*SerializedLevel
	once      sync.Once
	loadErr   error
}

// Levels 是全局关卡表实例。
var Levels = &LevelTable{}

// Load 从磁盘或内嵌数据加载关卡表。
// 优先读磁盘（便于开发期热更），失败则回退到内嵌数据。
func (lt *LevelTable) Load(dataDir string) error {
	lt.once.Do(func() {
		var raw []byte
		p := dataDir + "/levels.json"
		if b, err := os.ReadFile(p); err == nil {
			raw = b
		} else {
			raw = embeddedLevels
		}

		var table LevelTableFile
		if err := json.Unmarshal(raw, &table); err != nil {
			lt.loadErr = fmt.Errorf("解析关卡表: %w", err)
			return
		}
		if table.Count != len(table.Levels) {
			lt.loadErr = fmt.Errorf("关卡表 count=%d 与实际关卡数 %d 不符", table.Count, len(table.Levels))
			return
		}

		// 解法表
		var sols SolutionTableFile
		sraw := embeddedSolutions
		if b, err := os.ReadFile(dataDir + "/solutions.json"); err == nil {
			sraw = b
		}
		if err := json.Unmarshal(sraw, &sols); err != nil {
			lt.loadErr = fmt.Errorf("解析解法表: %w", err)
			return
		}
		if sols.Version != table.Version {
			lt.loadErr = fmt.Errorf("关卡表版本 %s 与解法表版本 %s 不一致", table.Version, sols.Version)
			return
		}

		lt.table = &table
		lt.byIdx = make(map[int]*SerializedLevel, len(table.Levels))
		for i := range table.Levels {
			lt.byIdx[table.Levels[i].Idx] = &table.Levels[i]
		}
		lt.solutions = make(map[int][]int, len(sols.Levels))
		for _, s := range sols.Levels {
			lt.solutions[s.Idx] = s.Solution
		}
	})
	return lt.loadErr
}

// Version 返回关卡表版本号。
func (lt *LevelTable) Version() string {
	if lt.table == nil {
		return ""
	}
	return lt.table.Version
}

// Count 返回关卡总数。
func (lt *LevelTable) Count() int {
	if lt.table == nil {
		return 0
	}
	return lt.table.Count
}

// Get 按关号取关卡，不存在返回 nil。
func (lt *LevelTable) Get(idx int) *SerializedLevel {
	return lt.byIdx[idx]
}

// Range 返回 [from, from+count) 区间的关卡（自动钳制到有效范围）。
func (lt *LevelTable) Range(from, count int) []SerializedLevel {
	if lt.table == nil || count <= 0 {
		return nil
	}
	if from < 1 {
		from = 1
	}
	if from > lt.table.Count {
		return nil
	}
	end := from + count
	if end > lt.table.Count+1 {
		end = lt.table.Count + 1
	}
	out := make([]SerializedLevel, 0, end-from)
	for i := from; i < end; i++ {
		out = append(out, lt.byIdx[i])
	}
	return out
}

// Solution 返回某关的解法（仅服务端内部使用，禁止外泄）。
func (lt *LevelTable) Solution(idx int) []int {
	return lt.solutions[idx]
}

// SelfCheck 启动自检：抽检若干关，验证解法与关卡一致。
func (lt *LevelTable) SelfCheck(sampleCount int) error {
	if lt.table == nil {
		return fmt.Errorf("关卡表未加载")
	}
	if lt.table.Count == 0 {
		return fmt.Errorf("关卡表为空")
	}
	step := lt.table.Count / sampleCount
	if step < 1 {
		step = 1
	}
	checked := 0
	for i := 1; i <= lt.table.Count; i += step {
		lv := lt.byIdx[i]
		if lv == nil {
			return fmt.Errorf("第 %d 关缺失", i)
		}
		sol := lt.solutions[i]
		if len(sol) == 0 {
			return fmt.Errorf("第 %d 关缺少解法", i)
		}
		if len(sol)-1 != lv.Shortest {
			return fmt.Errorf("第 %d 关解法步数 %d 与 shortest %d 不一致", i, len(sol)-1, lv.Shortest)
		}
		if sol[0] != indexOfType(lv, 3) || sol[len(sol)-1] != indexOfType(lv, 4) {
			return fmt.Errorf("第 %d 关解法起止点不正确", i)
		}
		checked++
	}
	if checked == 0 {
		return fmt.Errorf("自检未覆盖任何关卡")
	}
	return nil
}

// indexOfType 在 base64 网格中查找指定类型的格子索引。仅用于自检。
func indexOfType(lv *SerializedLevel, typ byte) int {
	raw, err := decodeGridBase64(lv.Grid)
	if err != nil {
		return -1
	}
	for i, b := range raw {
		if (b>>1)&0b111 == typ {
			return i
		}
	}
	return -1
}
```

### 9.2 写 `internal/games/line/codec.go`（Go 侧网格解码，用于成绩校验）

```go
package line

import (
	"encoding/base64"
	"fmt"
)

// 格子类型。与 TypeScript 侧保持一致。
const (
	CellBlock  = 0
	CellBrick  = 1
	CellOneway = 2
	CellStart  = 3
	CellEnd    = 4
)

// 方向。与 TypeScript 侧保持一致。
const (
	DirUp    = 0
	DirDown  = 1
	DirLeft  = 2
	DirRight = 3
)

// Cell 是解码后的单个格子。
type Cell struct {
	Type byte
	Dir  byte
}

// decodeGridBase64 解码 base64 网格为格子数组。
func decodeGridBase64(b64 string) ([]Cell, error) {
	raw, err := base64.StdEncoding.DecodeString(b64)
	if err != nil {
		return nil, fmt.Errorf("base64 解码: %w", err)
	}
	out := make([]Cell, len(raw))
	for i, b := range raw {
		isOneway := (b >> 7) & 1
		dir := (b >> 4) & 0b111
		typ := (b >> 1) & 0b111
		c := Cell{Type: typ, Dir: dir}
		if isOneway == 0 {
			c.Type = typ
		} else {
			c.Type = CellOneway
		}
		out[i] = c
	}
	return out, nil
}
```

package line

import (
	"strconv"
	"time"

	"github.com/gin-gonic/gin"

	"github.com/tupss/f-game/server/internal/core"
)

// ConfigHandler 返回关卡版本、皮肤表与广告配置。
func ConfigHandler(skins []SkinConfig, adID string) gin.HandlerFunc {
	return func(c *gin.Context) {
		core.OK(c, gin.H{
			"levelVersion": Levels.Version(),
			"levelCount":   Levels.Count(),
			"skins":        skins,
			"ad": gin.H{
				"rewardedVideoId": adID,
				"interstitialId":  "",
			},
			"features": gin.H{
				"hint":           true,
				"skip":           true,
				"shareChallenge": true,
			},
			"serverTime": time.Now().UnixMilli(),
		})
	}
}

// LevelsHandler 分批下发关卡表。
func LevelsHandler() gin.HandlerFunc {
	return func(c *gin.Context) {
		from, err := strconv.Atoi(c.DefaultQuery("from", "1"))
		if err != nil || from < 1 {
			core.Fail(c, core.CodeBadRequest, "from 参数非法")
			return
		}
		count, err := strconv.Atoi(c.DefaultQuery("count", "50"))
		if err != nil || count < 1 || count > 100 {
			core.Fail(c, core.CodeBadRequest, "count 参数需在 1..100 之间")
			return
		}
		levels := Levels.Range(from, count)
		core.OK(c, gin.H{
			"version": Levels.Version(),
			"from":    from,
			"levels":  levels,
		})
	}
}

// SkinConfig 描述一个皮肤的解锁条件。
type SkinConfig struct {
	ID          string `json:"id"`
	Name        string `json:"name"`
	Tier        string `json:"tier"`
	UnlockType  string `json:"unlockType"`
	UnlockValue int    `json:"unlockValue"`
}

package line

import (
	"testing"

func TestLoadAndSelfCheck(t *testing.T) {
	if err := Levels.Load("../../data"); err != nil {
		t.Fatalf("加载关卡表失败: %v", err)
	}
	if Levels.Count() != 1000 {
		t.Fatalf("期望 1000 关，实际 %d", Levels.Count())
	}
	if err := Levels.SelfCheck(50); err != nil {
		t.Fatalf("自检失败: %v", err)
	}
}

func TestRange(t *testing.T) {
	if err := Levels.Load("../../data"); err != nil {
		t.Fatalf("加载失败: %v", err)
	}
	got := Levels.Range(1, 50)
	if len(got) != 50 {
		t.Fatalf("期望 50 关，实际 %d", len(got))
	}
	if got[0].Idx != 1 {
		t.Fatalf("首关关号应为 1，实际 %d", got[0].Idx)
	}
	// 越界钳制
	if len(Levels.Range(990, 50)) != 11 {
		t.Fatalf("末尾区间应返回 11 关")
	}
	if Levels.Range(2000, 50) != nil {
		t.Fatalf("超出范围的请求应返回 nil")
	}
}

func TestDecodeGrid(t *testing.T) {
	if err := Levels.Load("../../data"); err != nil {
		t.Fatalf("加载失败: %v", err)
	}
	lv := Levels.Get(1)
	if lv == nil {
		t.Fatal("第 1 关不存在")
	}
	cells, err := decodeGridBase64(lv.Grid)
	if err != nil {
		t.Fatalf("解码失败: %v", err)
	}
	if len(cells) != lv.W*lv.H {
		t.Fatalf("格子数应为 %d，实际 %d", lv.W*lv.H, len(cells))
	}

## 坑
- `//go:embed` 要求文件**存在且在包目录内** → `internal/games/line/levels.json` 需要在 `internal/games/line/` 下。**若用磁盘加载则 embed 是备份**，但 `go:embed` 编译期要求文件存在 → **把 json 复制到 `internal/games/line/` 目录**，或改用 `//go:embed data/*.json` 并调整路径
  > **推荐做法**：把 `levels.json` 和 `solutions.json` 放在 `internal/games/line/` 下，用 `//go:embed levels.json`；`server/data/` 只作为开发期热更的备份副本
- `sync.Once` 的 `loadErr` 只会赋值一次 → 加载失败后无法重试（测试中会踩坑）。→ 若需要重试，把 `once` 换成普通判断
- `Range` 的区间是**闭区间 [from, from+count-1]**
- `SelfCheck` 里 `indexOfType` 解码 base64 后按 `(b>>1)&0b111` 判断类型 → 与 TS 侧编码一致
