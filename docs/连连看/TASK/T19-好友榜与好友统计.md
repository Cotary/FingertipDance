## 步骤

### 19.1 写 `src/platform/openData.ts`（主域侧）

```ts
import { platform } from './index';

/**
 * 开放数据域写入的玩家档案。
 * 好友榜与好友进度面板都读这个 KV。
 */
export interface FriendProfile {
  maxLevel: number;
  totalTimeMs: number;
  totalSteps: number;
  skinCount: number;
  nickname: string;
  avatar: string;
  updatedAt: number;
}

/** 排行榜分数公式。与服务端 ComputeScore 保持一致。 */
export function computeRankScore(maxLevel: number, totalTimeMs: number): number {
  const seconds = Math.min(999999, Math.floor(totalTimeMs / 1000));
  return maxLevel * 1000000 - seconds;
}

/**
 * 把自己的进度写入开放数据域，好友才能看到。
 */
export function publishProfile(p: FriendProfile): void {
  if (!platform.openDataAvailable) return;
  const wx = (globalThis as any).wx;
  if (!wx?.setUserCloudStorage) return;
  wx.setUserCloudStorage({
    KVDataList: [{
      key: 'line_profile',
      value: JSON.stringify(p),
    }],
    fail: (err: any) => console.warn('写入开放数据域失败', err),
  });
}

/**
 * 获取开放数据域的 canvas（用于把好友榜画到主域）。
 */
export function getOpenDataCanvas(): { canvas: any; width: number; height: number } | null {
  if (!platform.openDataAvailable) return null;
  const wx = (globalThis as any).wx;
  if (!wx?.getOpenDataContext) return null;
  const ctx = wx.getOpenDataContext();
  if (!ctx?.canvas) return null;
  return { canvas: ctx.canvas, width: ctx.canvas.width, height: ctx.canvas.height };
}
```

### 19.2 写 `games/line-wx/openDataContext/index.js`

**这是独立 JS 环境**，只能用 `wx.*`，不能引用主域任何东西。

```javascript
// 开放数据域：绘制好友榜
const { sharedCanvas, wx } = wx.getOpenDataContext();

// 好友榜布局（设计坐标）
const CELL_W = 90;
const CELL_H = 120;
const COLS = 3;
const ROWS = 4;

// 绘制单个好友卡片
function drawFriend(ctx, friend, x, y, size, rank) {
  // 底板
  ctx.fillStyle = rank <= 3 ? '#3A4460' : '#262D42';
  roundRect(ctx, x, y, size, size, 14);
  ctx.fill();

  // 头像
  if (friend.avatar) {
    const img = wx.createImage();
    img.src = friend.avatar;
    img.onload = function () {
      ctx.save();
      roundRect(ctx, x + size * 0.22, y + size * 0.12, size * 0.56, size * 0.56, size * 0.28);
      ctx.clip();
      ctx.drawImage(img, x + size * 0.22, y + size * 0.12, size * 0.56, size * 0.56);
      ctx.restore();
    };
  }

  // 昵称
  ctx.fillStyle = '#E8ECF5';
  ctx.font = '20px sans-serif';
  ctx.textAlign = 'center';
  const name = friend.nickname ? friend.nickname.slice(0, 4) : '玩家';
  ctx.fillText(name, x + size / 2, y + size * 0.78);

  // 关卡数
  ctx.fillStyle = '#F2C14E';
  ctx.font = 'bold 22px sans-serif';
  ctx.fillText('第' + friend.maxLevel + '关', x + size / 2, y + size * 0.93);
}

// 画圆角矩形
function roundRect(ctx, x, y, w, h, r) {
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.arcTo(x + w, y, x + w, y + h, r);
  ctx.arcTo(x + w, y + h, x, y + h, r);
  ctx.arcTo(x, y + h, x, y, r);
  ctx.arcTo(x, y, x + w, y, r);
  ctx.closePath();
}

// 主循环
function render() {
  const ctx = sharedCanvas.getContext('2d');
  ctx.clearRect(0, 0, sharedCanvas.width, sharedCanvas.height);
  ctx.fillStyle = '#1E2436';
  ctx.fillRect(0, 0, sharedCanvas.width, sharedCanvas.height);

  wx.getFriendCloudStorage({
    keyList: ['line_profile'],
    success: function (res) {
      const list = (res.data || []).filter(function (item) {
        return item.KVDataList && item.KVDataList.length > 0;
      }).map(function (item) {
        const kv = item.KVDataList[0];
        let p = {};
        try { p = JSON.parse(kv.value); } catch (e) { p = {}; }
        return {
          openid: item.openid,
          nickname: p.nickname || '',
          avatar: p.avatar || '',
          maxLevel: p.maxLevel || 0,
          totalTimeMs: p.totalTimeMs || 0,
          skinCount: p.skinCount || 0,
        };
      });

      // 按关卡数排序
      list.sort(function (a, b) {
        if (b.maxLevel !== a.maxLevel) return b.maxLevel - a.maxLevel;
        return a.totalTimeMs - b.totalTimeMs;
      });

      // 绘制
      const size = CELL_W;
      for (let i = 0; i < list.length && i < COLS * ROWS; i++) {
        const col = i % COLS;
        const row = Math.floor(i / COLS);
        drawFriend(ctx, list[i], col * (size + 12) + 12, row * (size + 12) + 12, size, i + 1);
      }

      if (list.length === 0) {
        ctx.fillStyle = '#8A94AD';
        ctx.font = '24px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText('还没有好友数据，快去分享给好友吧', sharedCanvas.width / 2, sharedCanvas.height / 2);
      }
    },
    fail: function () {
      console.error('读取好友数据失败');
    },
  });
}

render();
// 定时刷新（好友榜数据不会自动更新）
setInterval(render, 3000);
```

### 19.3 写 `src/ui/friends.ts`（好友榜页面 + 统计面板）

```ts
import { DESIGN_W, DESIGN_H } from '../render/layout';
import { drawButton, hitButton, COLORS, type Button } from './hud';
import { roundRect } from '../render/board';
import { getOpenDataCanvas, type FriendProfile } from '../platform/openData';

/** 好友条目（服务端版本）。 */
export interface FriendItem {
  uid: number;
  nickname: string;
  avatar: string;
  maxLevel: number;
  totalTimeMs: number;
  skinCount: number;
  isMutual: boolean;
}

/** 好友页状态。 */
export interface FriendsPageState {
  /** 服务端好友列表 */
  friends: FriendItem[];
  /** 统计 */
  stats: { friendCount: number; avgLevel: number; maxLevel: number };
  /** 我的名次（1-based），0 = 未上榜 */
  myRank: number;
  /** 是否正在加载开放数据域 */
  loadingOpenData: boolean;
}

/**
 * 绘制好友页。
 */
export function drawFriendsPage(ctx: CanvasRenderingContext2D, st: FriendsPageState): Button[] {
  ctx.fillStyle = COLORS.bg;
  ctx.fillRect(0, 0, DESIGN_W, DESIGN_H);

  ctx.fillStyle = COLORS.text;
  ctx.font = '700 38px system-ui, sans-serif';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'top';
  ctx.fillText('好友', DESIGN_W / 2, 60);

  // ── 统计面板（对应"好友关卡数量统计"需求）──
  const panelY = 120;
  ctx.fillStyle = COLORS.panel;
  roundRect(ctx, 40, panelY, DESIGN_W - 80, 180, 16);
  ctx.fill();

  const myLevel = 0; // 由调用方传入更好，这里用占位
  const statItems = [
    { label: '好友数', value: String(st.stats.friendCount) },
    { label: '好友平均进度', value: `${st.stats.avgLevel} 关` },
    { label: '好友最高进度', value: `${st.stats.maxLevel} 关` },
    { label: '我的名次', value: st.myRank > 0 ? `第 ${st.myRank} 名` : '未上榜' },
  ];
  statItems.forEach((s, i) => {
    const col = i % 2, row = Math.floor(i / 2);
    const x = 90 + col * 300;
    const y = panelY + 30 + row * 70;
    ctx.fillStyle = COLORS.subText;
    ctx.font = '22px system-ui, sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText(s.label, x, y);
    ctx.fillStyle = COLORS.yellow;
    ctx.font = '600 32px system-ui, sans-serif';
    ctx.fillText(s.value, x, y + 28);
  });

  // ── 好友榜（开放数据域）──
  const boardY = panelY + 210;
  ctx.fillStyle = COLORS.subText;
  ctx.font = '24px system-ui, sans-serif';
  ctx.textAlign = 'left';
  ctx.fillText('好友榜', 40, boardY);
  if (st.loadingOpenData) {
    ctx.fillStyle = COLORS.subText;
    ctx.textAlign = 'right';
    ctx.fillText('加载中...', DESIGN_W - 40, boardY);
  }

  const od = getOpenDataCanvas();
  if (od) {
    // 开放数据域的 canvas 尺寸是它自己的，需要缩放绘制
    const boardW = DESIGN_W - 80;
    const boardH = 420;
    try {
      ctx.drawImage(od.canvas, 40, boardY + 40, boardW, boardH);
    } catch {
      // 微信端首帧可能还没画好
    }
  } else {
    // Web 版用服务端好友列表兜底绘制
    ctx.fillStyle = COLORS.panel;
    roundRect(ctx, 40, boardY + 40, DESIGN_W - 80, 420, 16);
    ctx.fill();
    ctx.fillStyle = COLORS.text;
    ctx.textAlign = 'center';
    ctx.font = '26px system-ui, sans-serif';
    const list = st.friends.slice(0, 12);
    if (list.length === 0) {
      ctx.fillStyle = COLORS.subText;
      ctx.fillText('还没有好友，分享给好友一起来玩吧', DESIGN_W / 2, boardY + 250);
    }
    list.forEach((f, i) => {
      const y = boardY + 90 + i * 46;
      ctx.textAlign = 'left';
      ctx.fillStyle = COLORS.text;
      ctx.font = '24px system-ui, sans-serif';
      ctx.fillText(`${i + 1}. ${f.nickname || '玩家'}`, 70, y);
      ctx.textAlign = 'right';
      ctx.fillStyle = COLORS.yellow;
      ctx.fillText(`第 ${f.maxLevel} 关`, DESIGN_W - 70, y);
    });
  }

  // 返回按钮
  const back: Button = { x: 40, y: DESIGN_H - 140, w: 160, h: 72, label: '返回', kind: 'secondary', id: 'back' };
  drawButton(ctx, back, false);
  return [back];
}
```

### 19.4 在 `src/main.ts` 接入好友页

```ts
import { drawFriendsPage, type FriendsPageState, type FriendItem } from './ui/friends';
import { publishProfile } from './platform/openData';
import { fetchFriends } from './net/api';

let friendsPage: FriendsPageState = {
  friends: [],
  stats: { friendCount: 0, avgLevel: 0, maxLevel: 0 },
  myRank: 0,
  loadingOpenData: true,
};
let friendsButtons: Button[] = [];

/** 打开好友页。 */
async function openFriends() {
  phase = 'friends';
  friendsPage.loadingOpenData = true;
  reportEvent('skin_page_view', {});   // 修正为 leaderboard_view
  reportEvent('leaderboard_view', { scope: 'friends', my_rank: friendsPage.myRank });

  // 服务端好友列表（Web 兜底 + 统计）
  try {
    const res = await fetchFriends();
    friendsPage.friends = res.list as FriendItem[];
    friendsPage.stats = res.stats;
  } catch { /* 忽略 */ }

  // 发布自己的进度到开放数据域
  publishProfile({
    maxLevel: progress.current.maxLevel,
    totalTimeMs: progress.current.totalSteps * 1000,
    totalSteps: progress.current.totalSteps,
    skinCount: progress.current.unlockedSkins.length,
    nickname: '玩家',
    avatar: '',
    updatedAt: Date.now(),
  });

  // 开放数据域首帧需要时间
  setTimeout(() => { friendsPage.loadingOpenData = false; }, 500);
}
```

- `wx.setUserCloudStorage` 需要在登录后调用
- 开放数据域首帧是异步的 → 显示"加载中"

---

## 坑
- **开放数据域是独立环境** → 不能 import 主域任何模块，字体/绘制代码要复制一份
- **`sharedCanvas` 尺寸固定** → 主域 `drawImage` 时要缩放到目标矩形
- **好友榜数据不会自动刷新** → 开放数据域里 `setInterval(render, 3000)`
- **`wx.getFriendCloudStorage` 只返回互为好友的用户** → 这就是微信的规则，不要试图绕过
- **服务端好友列表与开放数据域好友不一致** → 两个都显示：开放数据域画榜，服务端算统计
- **`publishProfile` 里的 `totalTimeMs`** → 用 `totalSteps * 1000` 近似即可（或从服务端取真实值）
