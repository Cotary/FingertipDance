## 步骤

### 4.1 写 `src/core/curve.ts`

```ts
/** 单调插值。 */
function lerp(a: number, b: number, t: number): number {
  return a + (b - a) * t;
}
/** 缓入缓出，让段内曲线不平直。 */
function easeInOut(t: number): number {
  return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
}
/** 把值钳制到 [0,1]。 */
function clamp01(v: number): number {
  return v < 0 ? 0 : v > 1 ? 1 : v;
}

/** 一段难度配置。 */
interface Segment {
  /** 该段起始关号（含） */
  from: number;
  /** 该段结束关号（含） */
  to: number;
  /** 段内起始尺寸 */
  w: number; h: number;
  /** 段内结束尺寸 */
  w2: number; h2: number;
  /** 阻挡块数量区间 */
  blocks: [number, number];
  /** 单向块数量区间 */
  oneways: [number, number];
  /** 最少转弯数 */
  minTurns: number;
  /** 开放面积比下限 */
  openRatio: number;
}

/** 分段表。顺序必须按 from 升序。 */
export const SEGMENTS: Segment[] = [
  { from: 1,   to: 10,   w: 6, h: 6, w2: 6, h2: 6,  blocks: [0, 3],   oneways: [0, 0],  minTurns: 1, openRatio: 0.78 },
  { from: 11,  to: 30,   w: 7, h: 7, w2: 7, h2: 7,  blocks: [6, 12],  oneways: [0, 2],  minTurns: 2, openRatio: 0.72 },
  { from: 31,  to: 60,   w: 8, h: 8, w2: 8, h2: 8,  blocks: [12, 20], oneways: [3, 6],  minTurns: 3, openRatio: 0.66 },
  { from: 61,  to: 150,  w: 9, h: 9, w2: 9, h2: 9,  blocks: [20, 30], oneways: [6, 10], minTurns: 4, openRatio: 0.60 },
  { from: 151, to: 400,  w: 9, h: 9, w2: 9, h2: 9,  blocks: [28, 38], oneways: [8, 14], minTurns: 5, openRatio: 0.56 },
  { from: 401, to: 700,  w: 10, h: 10, w2: 10, h2: 10, blocks: [35, 48], oneways: [10, 16], minTurns: 6, openRatio: 0.52 },
  { from: 701, to: 1000, w: 10, h: 10, w2: 11, h2: 11, blocks: [45, 60], oneways: [12, 20], minTurns: 7, openRatio: 0.48 },
];

/** 生成参数包。 */
export interface CurveParams {
  /** 棋盘列数 */
  w: number;
  /** 棋盘行数 */
  h: number;
  /** 阻挡块数量 */
  blocks: number;
  /** 单向块数量 */
  oneways: number;
  /** 正解路径最少转弯数 */
  minTurns: number;
  /** 目标开放面积比下限 */
  openRatio: number;
  /** 正解路径的最小长度（步数） */
  minPathLen: number;
  /** 分叉抑制强度 0..1，越大越追求唯一解 */
  branchPenalty: number;
}

/** 找到关号所属的分段。 */
function segmentOf(idx: number): Segment {
  for (const s of SEGMENTS) if (idx >= s.from && idx <= s.to) return s;
  return SEGMENTS[SEGMENTS.length - 1];
}

/**
 * 计算关卡的生成参数。纯函数，不依赖随机数。
 * @param idx 关号，1 起
 */
export function curveFor(idx: number): CurveParams {
  const seg = segmentOf(idx);
  const span = seg.to - seg.from;
  const t = span <= 0 ? 0 : (idx - seg.from) / span;
  const e = easeInOut(t);

  // 大关（每 100 关）：相对同段略微放宽，转弯更多 → "绕但不死"
  const isBig = idx % 100 === 0;
  const isPeak = idx % 10 === 0;

  const w = Math.round(lerp(seg.w, seg.w2, e));
  const h = Math.round(lerp(seg.h, seg.h2, e));

  let blocks = Math.round(lerp(seg.blocks[0], seg.blocks[1], e));
  let oneways = Math.round(lerp(seg.oneways[0], seg.oneways[1], e));
  let openRatio = lerp(seg.openRatio, seg.openRatio - 0.04, e);

  if (isBig) {
    blocks = Math.max(0, blocks - 6);
    openRatio = Math.min(0.9, openRatio + 0.04);
  }

  const minTurns = seg.minTurns + (isBig ? 1 : 0);
  const minPathLen = Math.max(4, Math.round(Math.max(w, h) * 1.4));

  return {
    w, h, blocks, oneways, minTurns, openRatio, minPathLen,
    branchPenalty: clamp01(0.35 + (idx / 1000) * 0.5),
  };
}

/**
 * 目标难度分（0..100）。
 * 1~10 关从 5 升到 15，11~1000 从 15 升到 92；
 * 每 10 关加 8 分小高峰；每 100 关额外 +4。
 */
export function targetDiff(idx: number): number {
  let base: number;
  if (idx <= 10) {
    base = lerp(5, 15, (idx - 1) / 9);
  } else {
    base = lerp(15, 92, Math.min(1, (idx - 10) / 990));
  }
  if (idx % 10 === 0) base += 8;
  if (idx % 100 === 0) base += 4;
  return Math.max(0, Math.min(100, Math.round(base)));
}

/**
 * 依据关卡实际构成计算难度分。
 * 公式见 02-关卡格式与算法规格.md 第 4.2 节。
 */
export function computeDiff(input: {
  w: number; h: number;
  blocks: number; oneways: number;
  turns: number;
  /** 最短路条数，最小 1 */
  pathCount: number;
}): number {
  const { w, h, blocks, oneways, turns, pathCount } = input;
  const area = w * h;
  const uniqueness = pathCount <= 1 ? 1 : clamp01(1 - Math.log2(pathCount) / 6);

  const raw =
    0.30 * clamp01(blocks / (area * 0.45)) +
    0.15 * clamp01(oneways / (area * 0.25)) +
    0.25 * clamp01(turns / 8) +
    0.20 * clamp01((1 - input.openRatio ?? 0) / 0.55) +
    0.10 * clamp01(1 - uniqueness);

  return Math.max(0, Math.min(100, Math.round(raw * 100)));
}
```

> **注意上面 `computeDiff` 的 `openRatio` 参数缺失** —— 请在函数签名里补上：

>   blocks: number; oneways: number;

>   pathCount: number;
>   /** 实际开放面积比 */
>   openRatio: number;
> }): number
> ```

### 4.2 写测试 `games/line/test/curve.test.ts`

```ts
import { describe, it, expect } from 'vitest';
import { curveFor, targetDiff, computeDiff, SEGMENTS } from '../src/core/curve';

describe('curve', () => {
  it('分段覆盖 1..1000 无空隙', () => {
    let prev = 0;
    for (const s of SEGMENTS) {
      expect(s.from).toBe(prev + 1);
      prev = s.to;
    }
    expect(prev).toBe(1000);
  });

  it('尺寸随关号单调不减', () => {
    let prevArea = 0;
    for (let idx = 1; idx <= 1000; idx += 7) {
      const p = curveFor(idx);
      const area = p.w * p.h;
      expect(area).toBeGreaterThanOrEqual(prevArea);
      prevArea = area;
    }
  });

  it('阻挡块数量随关号不减', () => {
    let prev = -1;
    for (let idx = 1; idx <= 1000; idx += 13) {
      const p = curveFor(idx);
      expect(p.blocks).toBeGreaterThanOrEqual(prev);
      prev = p.blocks;
    }
  });

  it('开放面积比随关号递减', () => {
    let prev = 2;
    for (let idx = 1; idx <= 1000; idx += 17) {
      const p = curveFor(idx);
      expect(p.openRatio).toBeLessThanOrEqual(prev + 1e-9);
      prev = p.openRatio;
    }
  });

  it('targetDiff 单调不减（含小高峰容差）', () => {
    for (let idx = 1; idx < 1000; idx++) {
      const a = targetDiff(idx);
      const b = targetDiff(idx + 1);
      // 小高峰后回落是允许的（最多回落 8）
      if (b < a) expect(a - b).toBeLessThanOrEqual(8);
    }
  });

  it('targetDiff 在 [0,100]', () => {
    for (let idx = 1; idx <= 1000; idx++) {
      const d = targetDiff(idx);
      expect(d).toBeGreaterThanOrEqual(0);
      expect(d).toBeLessThanOrEqual(100);
    }
  });

  it('大关比同段普通关更宽松', () => {
    const big = curveFor(200);
    const normal = curveFor(201);
    expect(big.blocks).toBeLessThanOrEqual(normal.blocks);
    expect(big.openRatio).toBeGreaterThanOrEqual(normal.openRatio);
  });

  it('computeDiff 唯一解比多解更难', () => {
    const base = { w: 9, h: 9, blocks: 20, oneways: 6, turns: 4, openRatio: 0.6 };
    const unique = computeDiff({ ...base, pathCount: 1 });
    const multi = computeDiff({ ...base, pathCount: 16 });
    expect(unique).toBeGreaterThan(multi);
  });

  it('computeDiff 在 [0,100]', () => {
    const d = computeDiff({ w: 11, h: 11, blocks: 60, oneways: 20, turns: 9, pathCount: 1, openRatio: 0.45 });
    expect(d).toBeGreaterThanOrEqual(0);
    expect(d).toBeLessThanOrEqual(100);
  });
});
```

## 坑
- `segments` 数组必须覆盖 1..1000 **无空隙无重叠**，否则 `segmentOf` 会落到 fallback
- 大关放宽（blocks-6）会让 200 关的 blocks 小于 199 关 → 上面「阻挡块数量随关号不减」的测试用 `idx += 13` 采样可能刚好跳过 → 若测试失败，把采样步长改成质数（如 11, 13, 17）并对大关做例外断言
- `computeDiff` 漏 `openRatio` 参数 → TS 会报错，务必补齐
