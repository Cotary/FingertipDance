### 12.1 写 `internal/modules/leaderboard/leaderboard.go`

```go
package leaderboard

import (
	"context"
	"database/sql"
	"fmt"
	"strconv"
	"time"

	"github.com/gin-gonic/gin"

	"github.com/tupss/f-game/server/internal/core"
)

// 排行榜 Redis key。
const (
	KeyGlobal = "rank:global"
	KeyWeekly = "rank:weekly"
)

// 分数换算：把「最高关卡 + 总用时」压成一个整数，避免浮点误差。
// 分数越大排名越靠前。
const scoreUnit = 1000000

// ComputeScore 计算排行榜分数。
// 主指标是最高关卡数，同关卡比总用时（用时少者靠前）。
func ComputeScore(maxLevel int, totalTimeMs int64) int64 {
	seconds := totalTimeMs / 1000
	if seconds > 999999 {
		seconds = 999999
	}
	return int64(maxLevel)*scoreUnit - seconds
}

// UpdateScore 更新某个用户的排行榜分数（只增不减）。
func UpdateScore(ctx context.Context, rdb *core.Redis, uid int64, maxLevel int, totalTimeMs int64) error {
	score := ComputeScore(maxLevel, totalTimeMs)
	member := strconv.FormatInt(uid, 10)
	// ZAddGT：只有更高分才更新
	return rdb.ZAddArgs(ctx, KeyGlobal, redis.ZAddArgs{
		GT:      true,
		Members: []redis.Z{{Score: float64(score), Member: member}},
	}).Err()
}

// 榜单条目。
type entry struct {
	Rank        int    `json:"rank"`
	UID        int64  `json:"uid"`
	Nickname   string `json:"nickname"`
	Avatar     string `json:"avatar"`
	MaxLevel   int    `json:"maxLevel"`
	TotalTimeMs int64 `json:"totalTimeMs"`
}

// GetHandler 返回总榜。
func GetHandler(db *core.DB, rdb *core.Redis) gin.HandlerFunc {
	return func(c *gin.Context) {
		ctx := c.Request.Context()
		uid := core.UIDFromContext(c)

		// 取 Top 100
		zs, err := rdb.ZRevRangeWithScores(ctx, KeyGlobal, 0, 99).Result()
		if err != nil {
			core.Fail(c, core.CodeInternal, "读取榜单失败: "+err.Error())
			return
		}

		list := make([]entry, 0, len(zs))
		uids := make([]int64, 0, len(zs))
		for _, z := range zs {
			member, _ := z.Member.(string)
			parsed, err := strconv.ParseInt(member, 10, 64)
			if err != nil {
				continue
			}
			uids = append(uids, parsed)
		}

		infoMap := loadUserInfo(db, uids)
		for i, z := range zs {
			member, _ := z.Member.(string)
			parsed, _ := strconv.ParseInt(member, 10, 64)
			e := entry{Rank: i + 1, UID: parsed}
			if info, ok := infoMap[parsed]; ok {
				e.Nickname = info.nickname
				e.Avatar = info.avatar
				e.MaxLevel = info.maxLevel
				e.TotalTimeMs = info.totalTimeMs
			}
			list = append(list, e)
		}

		// 我的排名
		myRank := int64(0)
		if r, err := rdb.ZRevRank(ctx, KeyGlobal, strconv.FormatInt(uid, 10)).Result(); err == nil {
			myRank = r + 1
		}

		total, _ := rdb.ZCard(ctx, KeyGlobal).Result()

		core.OK(c, gin.H{
			"list":   list,
			"myRank": myRank,
			"total":  total,
		})
	}
}

// userInfo 是榜单需要的用户信息子集。
type userInfo struct {
	nickname     string
	avatar       string
	maxLevel     int
	totalTimeMs  int64
}

// loadUserInfo 批量查询用户信息。
func loadUserInfo(db *core.DB, uids []int64) map[int64]userInfo {
	out := make(map[int64]userInfo, len(uids))
	if len(uids) == 0 {
		return out
	}
	// 构造 IN 查询
	placeholders := ""
	args := make([]interface{}, 0, len(uids))
	for i, id := range uids {
		if i > 0 {
			placeholders += ","
		}
		placeholders += "?"
		args = append(args, id)
	}
	q := "SELECT id, nickname, avatar, max_level, total_time_ms FROM users WHERE id IN (" + placeholders + ")"
	rows, err := db.Query(q, args...)
	if err != nil {
		return out
	}
	defer rows.Close()
	for rows.Next() {
		var id int64
		var u userInfo
		if err := rows.Scan(&id, &u.nickname, &u.avatar, &u.maxLevel, &u.totalTimeMs); err != nil {
			continue
		}
		out[id] = u
	}
	return out
}

// RefreshAll 重建整个榜单（定时任务用）。
func RefreshAll(ctx context.Context, db *core.DB, rdb *core.Redis) error {
	rows, err := db.Query("SELECT id, max_level, total_time_ms FROM users WHERE max_level > 0")
	if err != nil {
		return fmt.Errorf("查询用户: %w", err)
	}
	defer rows.Close()

	members := make([]redis.Z, 0, 256)
	for rows.Next() {
		var id int64
		var maxLevel int
		var totalTime int64
		if err := rows.Scan(&id, &maxLevel, &totalTime); err != nil {
			continue
		}
		members = append(members, redis.Z{
			Score:  float64(ComputeScore(maxLevel, totalTime)),
			Member: strconv.FormatInt(id, 10),
		})
	}
	if len(members) == 0 {
		return nil
	}
	// 分批写入
	for i := 0; i < len(members); i += 500 {
		end := i + 500
		if end > len(members) {
			end = len(members)
		}
		if err := rdb.ZAdd(ctx, KeyGlobal, members[i:end]...).Err(); err != nil {
			return fmt.Errorf("写入榜单: %w", err)
		}
	}
	return nil
}
```
> **需要额外 import**：`"github.com/redis/go-redis/v9"`（`redis.Z`、`redis.ZAddArgs`）。

### 12.2 排行榜每日重建

在 `cmd/http/main.go` 里加一个后台 goroutine：
```go
	// 每日凌晨重建总榜（保证与数据库一致）
	go func() {
		for {
			now := time.Now()
			next := time.Date(now.Year(), now.Month(), now.Day()+1, 4, 0, 0, 0, now.Location())
			select {
			case <-time.After(time.Until(next)):
				if err := leaderboard.RefreshAll(context.Background(), db, rdb); err != nil {
					logger.Error("重建榜单失败", zap.Error(err))
				} else {
					logger.Info("总榜已重建")
				}
			case <-ctx.Done():
				return
			}
		}
	}()
```

### 12.3 写 `internal/modules/friend/friend.go`

```go
package friend

import (
	"database/sql"
	"fmt"
	"time"

	"github.com/gin-gonic/gin"

	"github.com/tupss/f-game/server/internal/core"
)

// 好友状态。
const (
	StatusPending  = 0
	StatusAccepted = 1
	StatusBlocked  = 2
)

// 好友操作。
type actionReq struct {
	Action    string `json:"action"`
	FriendUID int64  `json:"friendUid"`
}

// 好友条目。
type FriendItem struct {
	UID         int64  `json:"uid"`
	Nickname    string `json:"nickname"`
	Avatar      string `json:"avatar"`
	MaxLevel    int    `json:"maxLevel"`
	TotalTimeMs int64  `json:"totalTimeMs"`
	SkinCount   int    `json:"skinCount"`
	IsMutual    bool   `json:"isMutual"`
}

// Handler 处理好友操作与查询。
func Handler(db *core.DB) gin.HandlerFunc {
	return func(c *gin.Context) {
		switch c.Request.Method {
		case http.MethodPost:
			handleAction(c, db)
		case http.MethodGet:
			handleList(c, db)
		default:
			core.Fail(c, core.CodeBadRequest, "不支持的方法")
		}
	}
}

// handleAction 处理申请/同意/删除/拉黑。
func handleAction(c *gin.Context, db *core.DB) {
	var req actionReq
	if err := c.ShouldBindJSON(&req); err != nil {
		core.Fail(c, core.CodeBadRequest, "参数错误: "+err.Error())
		return
	}
	uid := core.UIDFromContext(c)
	if req.FriendUID == 0 || req.FriendUID == uid {
		core.Fail(c, core.CodeBadRequest, "好友 ID 非法")
		return
	}
	now := time.Now()

	switch req.Action {
	case "apply":
		_, err := db.Exec(
			"INSERT INTO friends (uid, friend_uid, status, created_at) VALUES (?,?,?,?) "+
				"ON DUPLICATE KEY UPDATE status=VALUES(status)",
			uid, req.FriendUID, StatusPending, now,
		)
		if err != nil {
			core.Fail(c, core.CodeInternal, "发起申请失败: "+err.Error())
			return
		}
		// 对方也需要能看到这条申请
		_, _ = db.Exec(
			"INSERT INTO friends (uid, friend_uid, status, created_at) VALUES (?,?,?,?) "+
				"ON DUPLICATE KEY UPDATE status=VALUES(status)",
			req.FriendUID, uid, StatusPending, now,
		)

	case "accept":
		_, err := db.Exec(
			"UPDATE friends SET status=? WHERE (uid=? AND friend_uid=?) OR (uid=? AND friend_uid=?)",
			StatusAccepted, uid, req.FriendUID, req.FriendUID, uid,
		)
		if err != nil {
			core.Fail(c, core.CodeInternal, "同意失败: "+err.Error())
			return
		}

	case "remove":
		_, err := db.Exec(
			"DELETE FROM friends WHERE (uid=? AND friend_uid=?) OR (uid=? AND friend_uid=?)",
			uid, req.FriendUID, req.FriendUID, uid,
		)
		if err != nil {
			core.Fail(c, core.CodeInternal, "删除失败: "+err.Error())
			return
		}

	case "block":
		_, err := db.Exec(
			"UPDATE friends SET status=? WHERE uid=? AND friend_uid=?",
			StatusBlocked, uid, req.FriendUID,
		)
		if err != nil {
			core.Fail(c, core.CodeInternal, "拉黑失败: "+err.Error())
			return
		}

	case "unblock":
		_, err := db.Exec(
			"UPDATE friends SET status=? WHERE uid=? AND friend_uid=?",
			StatusPending, uid, req.FriendUID,
		)
		if err != nil {
			core.Fail(c, core.CodeInternal, "解除拉黑失败: "+err.Error())
			return
		}

	default:
		core.Fail(c, core.CodeBadRequest, "未知操作: "+req.Action)
		return
	}

	core.OK(c, gin.H{"ok": true})
}

// handleList 返回好友列表与进度统计。
func handleList(c *gin.Context, db *core.DB) {
	uid := core.UIDFromContext(c)
	rows, err := db.Query(
		"SELECT u.id, u.nickname, u.avatar, u.max_level, u.total_time_ms, "+
			"JSON_LENGTH(u.unlocked_skins) FROM friends f "+
			"JOIN users u ON u.id = f.friend_uid "+
			"WHERE f.uid = ? AND f.status = ?",
		uid, StatusAccepted,
	)
	if err != nil {
		core.Fail(c, core.CodeInternal, "查询好友失败: "+err.Error())
		return
	}
	defer rows.Close()

	list := make([]FriendItem, 0, 32)
	for rows.Next() {
		var it FriendItem
		var skinCount sql.NullInt64
		if err := rows.Scan(&it.UID, &it.Nickname, &it.Avatar, &it.MaxLevel, &it.TotalTimeMs, &skinCount); err != nil {
			continue
		}
		it.SkinCount = int(skinCount.Int64)
		it.IsMutual = true // 双方 accepted 时即为互关
		list = append(list, it)
	}

	// 统计：好友平均进度、最高进度好友
	avg := 0
	max := 0
	if len(list) > 0 {
		sum := 0
		for _, it := range list {
			sum += it.MaxLevel
			if it.MaxLevel > max {
				max = it.MaxLevel
			}
		}
		avg = sum / len(list)
	}

	core.OK(c, gin.H{
		"list": list,
		"stats": gin.H{
			"friendCount": len(list),
			"avgLevel":    avg,
			"maxLevel":    max,
		},
	})
}
```

### 12.4 写 Go 侧测试 `internal/modules/leaderboard/leaderboard_test.go`

package leaderboard

import "testing"

func TestComputeScore(t *testing.T) {
	// 关卡数越多分数越高
	if ComputeScore(10, 5000) <= ComputeScore(9, 1000) {
		t.Fatal("关卡数应主导分数")
	}
	// 同关卡数时用时少者分数高
	if ComputeScore(10, 3000) <= ComputeScore(10, 5000) {
		t.Fatal("同关卡数应按用时少者靠前")
	}
	// 用时被钳制，不会溢出到影响关卡维度
	s := ComputeScore(1, 999999999)
	if s <= 0 {
		t.Fatalf("分数不应为负: %d", s)
	}
	if s >= ComputeScore(2, 0) {
		t.Fatal("钳制后 1 关的分数仍应低于 2 关")
	}
}

## 坑
- `redis.ZAddArgs{GT: true}` 只在分数更高时更新 → 防止重放旧请求把分数降回去
- `ComputeScore` 用整数而非浮点 → 避免 Redis ZSet 的 float64 精度问题
- `ZRevRank` 返回 0-based 排名 → 响应里 +1
- `loadUserInfo` 的 IN 查询要防 SQL 注入 → 用占位符参数，**不要**字符串拼接 uid
- `friends` 表双向存两行 → 同意时用 `UPDATE ... WHERE (uid=? AND friend_uid=?) OR (...)` 一次更新两边
- 每日重建榜单的 goroutine 需要 `ctx` 可取消 → 否则服务关闭时不退出
