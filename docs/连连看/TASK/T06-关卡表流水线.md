### 6.1 写 `src/core/levelTable.ts`（序列化与反序列化）

```ts
import { encodeGrid, decodeGrid } from './codec';
import { CellType, type Cell, type Level } from './types';
import { flatten, unflatten } from './grid';

/** 关卡表（公开）。 */
export interface LevelTableFile {
  version: string;
  count: number;
  generatedAt: string;
  levels: SerializedLevel[];
}

/** 关卡表中每关的序列化形式。 */
export interface SerializedLevel {
  idx: number;
  seed: number;
  w: number;
  h: number;
  grid: string;
  shortest: number;
  turns: number;
  blocks: number;
  oneways: number;
  diff: number;
  unlockSkin: string | null;
}

/** 私密解法表。 */
export interface SolutionTableFile {
  version: string;
  levels: Array<{ idx: number; solution: number[]; solutionSteps: number; turns: number; diff: number }>;
}

/** 把关卡序列化为可 JSON 的对象。 */
export function serializeLevel(lv: Level, unlockSkin: string | null): SerializedLevel {
  return {
    idx: lv.idx, seed: lv.seed, w: lv.w, h: lv.h,
    grid: encodeGrid(flatten(lv.grid), lv.w, lv.h),
    shortest: lv.shortest, turns: lv.turns,
    blocks: lv.blocks, oneways: lv.oneways, diff: lv.diff,
    unlockSkin,
  };
}

/** 从序列化对象还原关卡。 */
export function deserializeLevel(s: SerializedLevel): Level {
  const flat = decodeGrid(s.grid, s.w, s.h);
  return {
    idx: s.idx, seed: s.seed, w: s.w, h: s.h,
    grid: unflatten(flat, s.w, s.h),
    shortest: s.shortest, turns: s.turns,
    blocks: s.blocks, oneways: s.oneways, diff: s.diff,
    unlockSkin: s.unlockSkin,
  };
}

/** 计算内容哈希（版本号）。 */
export async function computeVersion(content: string): Promise<string> {
  // 浏览器与 Node 都能用
  const subtle = (globalThis as any).crypto?.subtle;
  if (subtle) {
    const buf = await subtle.digest('SHA-256', new TextEncoder().encode(content));
    const hex = Array.from(new Uint8Array(buf)).map((b) => b.toString(16).padStart(2, '0')).join('');
    return `sha256:${hex.slice(0, 12)}`;
  }
  // Node 环境回退
  const { createHash } = await import('node:crypto');
  return `sha256:${createHash('sha256').update(content).digest('hex').slice(0, 12)}`;
}
```

### 6.2 写 `tools/level-gen.ts`

```ts
import { writeFileSync, readFileSync, existsSync, mkdirSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { generateLevel } from '../src/core/generator';
import { targetDiff } from '../src/core/curve';
import { solve, isPathLegal } from '../src/core/solver';
import { serializeLevel, computeVersion, type LevelTableFile, type SolutionTableFile, type SerializedLevel } from '../src/core/levelTable';

interface Args {
  count: number; seed: number; out: string; solutions: string; start: number;
}

function parseArgs(): Args {
  const a = process.argv.slice(2);
  const get = (k: string, d: string) => {
    const i = a.indexOf(`--${k}`);
    return i >= 0 && a[i + 1] ? a[i + 1] : d;
  };
  return {
    count: parseInt(get('count', '1000'), 10),
    seed: parseInt(get('seed', '20261101'), 10),
    out: resolve(get('out', 'data/levels.json')),
    solutions: resolve(get('solutions', 'data/solutions.json')),
    start: parseInt(get('start', '1'), 10),
  };
}

async function main() {
  const args = parseArgs();
  console.log(`生成关卡：count=${args.count} seed=${args.seed} start=${args.start}`);

  const existing: SerializedLevel[] = [];
  if (args.start > 1 && existsSync(args.out)) {
    const old = JSON.parse(readFileSync(args.out, 'utf8')) as LevelTableFile;
    existing.push(...old.levels.filter((l) => l.idx < args.start));
    console.log(`保留已有 ${existing.length} 关`);
  }

  const levels: SerializedLevel[] = [];
  const solutions: SolutionTableFile['levels'] = [];
  const fingerprints = new Set<string>();

  for (let i = 0; i < args.count; i++) {
    const idx = args.start + i;
    const t0 = Date.now();
    let lv = generateLevel(idx, args.seed);

    // 去重：布局指纹
    let fp = '';
    for (let attempt = 0; attempt < 5; attempt++) {
      fp = JSON.stringify(lv.grid);
      if (!fingerprints.has(fp)) break;
      lv = generateLevel(idx, args.seed + attempt * 7919);
    }
    if (fingerprints.has(fp)) {
      console.warn(`⚠ 第 ${idx} 关去重失败，跳过`);
      continue;
    }
    fingerprints.add(fp);

    // 独立二次校验（防生成器 bug）
    const res = solve(lv.grid);
    if (!res.solvable) {
      console.error(`✗ 第 ${idx} 关不可解，流水线中止`);
      process.exit(1);
    }
    if (!isPathLegal(lv.grid, res.paths[0])) {
      console.error(`✗ 第 ${idx} 关 canonical 路径非法，流水线中止`);
      process.exit(1);
    }

    const unlockSkin = skinForLevel(idx);
    levels.push(serializeLevel(lv, unlockSkin));
    solutions.push({
      idx,
      solution: res.paths[0],
      solutionSteps: res.steps,
      turns: lv.turns,
      diff: lv.diff,
    });

    const delta = Math.abs(lv.diff - targetDiff(idx));
    if (idx % 50 === 0 || delta > 12) {
      console.log(`[${String(levels.length).padStart(4)}/${args.count}] idx=${idx} diff=${lv.diff} target=${targetDiff(idx)} steps=${lv.shortest} ${Date.now() - t0}ms`);
    }
  }

  const allLevels = [...existing, ...levels].sort((a, b) => a.idx - b.idx);

  // 先写内容，算版本号，再回写
  const provisional = {
    version: '',
    count: allLevels.length,
    generatedAt: new Date().toISOString(),
    levels: allLevels,
  };
  const version = await computeVersion(JSON.stringify(provisional.levels));
  provisional.version = version;

  const solProvisional = { version: '', levels: solutions };
  solProvisional.version = version;

  mkdirSync(dirname(args.out), { recursive: true });
  writeFileSync(args.out, JSON.stringify(provisional), 'utf8');
  writeFileSync(args.solutions, JSON.stringify(solProvisional), 'utf8');

  console.log(`\n✓ 已写入 ${args.out}`);
  console.log(`✓ 已写入 ${args.solutions}`);
  console.log(`✓ version = ${version}，共 ${allLevels.length} 关`);

  // 写完后回读校验
  verify(args.out, args.solutions, version);
}

/** 皮肤解锁映射：见 04-美术规范.md 第 6.2 节。 */
function skinForLevel(idx: number): string | null {
  const map: Array<[number, string]> = [
    [3, 'graphite'], [8, 'slate'], [15, 'blush'], [25, 'amber'],
    [40, 'marble'], [60, 'brushed'], [90, 'ice'], [130, 'neon'],
    [200, 'gold'], [500, 'stardust'], [900, 'galaxy'],
  ];
  for (const [level, skin] of map) if (idx === level) return skin;
  return null;
}

/** 回读校验：解码每关并用解法验证。 */
function verify(levelsPath: string, solPath: string, version: string) {
  const lv = JSON.parse(readFileSync(levelsPath, 'utf8')) as LevelTableFile;
  const so = JSON.parse(readFileSync(solPath, 'utf8')) as SolutionTableFile;
  if (lv.version !== version || so.version !== version) {
    console.error('✗ 版本号不一致'); process.exit(1);
  }
  if (lv.count !== lv.levels.length) {
    console.error('✗ count 与实际关卡数不符'); process.exit(1);
  }
  const solMap = new Map(so.levels.map((s) => [s.idx, s]));
  let checked = 0;
  for (const s of lv.levels) {
    const sol = solMap.get(s.idx);
    if (!sol) continue;
    const { deserializeLevel } = require('../src/core/levelTable') as typeof import('../src/core/levelTable');
    const level = deserializeLevel(s);
    if (!isPathLegal(level.grid, sol.solution)) {
      console.error(`✗ 第 ${s.idx} 关解法非法`); process.exit(1);
    }
    checked++;
  }
  console.log(`✓ 回读校验通过（${checked} 关）`);
}

main().catch((e) => { console.error(e); process.exit(1); });
```
> **注意**：上面 `verify` 里的 `require` 在 ESM 下不可用。**改成顶部静态导入** `import { serializeLevel, computeVersion, deserializeLevel, ... } from '../src/core/levelTable';`，然后直接用 `deserializeLevel(s)`。

### 6.3 写 `tools/batch-audit.ts`

```ts
import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { deserializeLevel, type LevelTableFile } from '../src/core/levelTable';
import { solve } from '../src/core/solver';
import { targetDiff } from '../src/core/curve';

function arg(k: string, d: string) {
  const a = process.argv.slice(2);
  const i = a.indexOf(`--${k}`);
  return i >= 0 && a[i + 1] ? a[i + 1] : d;
}

function main() {
  const file = resolve(arg('file', 'data/levels.json'));
  const reportPath = resolve(arg('report', 'out/audit.md'));
  const table = JSON.parse(readFileSync(file, 'utf8')) as LevelTableFile;

  let unsolvable = 0, dupCount = 0, encodeFail = 0;
  const fps = new Set<string>();
  const buckets: Array<{ from: number; to: number; diffs: number[]; steps: number[]; turns: number[]; blockRate: number[]; uniqueRate: number[] }> = [];
  const anomalies: string[] = [];

  for (let i = 0; i < table.levels.length; i += 10) {
    buckets.push({ from: i + 1, to: Math.min(i + 10, table.levels.length), diffs: [], steps: [], turns: [], blockRate: [], uniqueRate: [] });
  }

  for (const s of table.levels) {
    const level = deserializeLevel(s);
    const fp = s.grid;
    if (fps.has(fp)) dupCount++;
    fps.add(fp);

    const res = solve(level.grid);
    if (!res.solvable) { unsolvable++; anomalies.push(`第 ${s.idx} 关不可解`); continue; }
    if (res.steps !== s.shortest) anomalies.push(`第 ${s.idx} 关 shortest 不一致（表=${s.shortest} 实=${res.steps}）`);

    const b = buckets[Math.floor((s.idx - 1) / 10)];
    b.diffs.push(s.diff);
    b.steps.push(s.shortest);
    b.turns.push(s.turns);
    b.blockRate.push(s.blocks / (s.w * s.h));
    b.uniqueRate.push(res.pathCount <= 1 ? 1 : 0);

    if (Math.abs(s.diff - targetDiff(s.idx)) > 12) {
      anomalies.push(`第 ${s.idx} 关难度偏离：diff=${s.diff} target=${targetDiff(s.idx)}`);
    }
    if (s.shortest < 8 && s.idx > 10) {
      anomalies.push(`第 ${s.idx} 关过于简单：shortest=${s.shortest}`);
    }
  }

  const avg = (a: number[]) => a.length ? a.reduce((x, y) => x + y, 0) / a.length : 0;
  const lines: string[] = [];
  lines.push(`# 关卡表体检报告`);
  lines.push(`\n文件：${file}`);
  lines.push(`版本：${table.version}`);
  lines.push(`关卡数：${table.count}`);
  lines.push(`\n## 总览`);
  lines.push(`- 不可解：${unsolvable}`);
  lines.push(`- 重复布局：${dupCount}`);
  lines.push(`- 编码往返失败：${encodeFail}`);
  lines.push(`- 异常项：${anomalies.length}`);

  lines.push(`\n## 每 10 关一档`);
  lines.push(`\n| 关卡 | 平均diff | 目标diff | 平均步数 | 平均转弯 | 平均阻挡率 | 唯一解占比 |`);
  lines.push(`|---|---|---|---|---|---|---|`);
  for (const b of buckets) {
    const td = targetDiff(b.from);
    lines.push(`| ${b.from}-${b.to} | ${avg(b.diffs).toFixed(1)} | ${td} | ${avg(b.steps).toFixed(1)} | ${avg(b.turns).toFixed(1)} | ${(avg(b.blockRate) * 100).toFixed(1)}% | ${(avg(b.uniqueRate) * 100).toFixed(0)}% |`);
  }

  // 直方图
  lines.push(`\n## diff 分布直方图`);
  const hist = new Array(10).fill(0);
  for (const s of table.levels) hist[Math.min(9, Math.floor(s.diff / 10))]++;
  for (let i = 0; i < 10; i++) {
    const bar = '█'.repeat(Math.round((hist[i] / table.count) * 100));
    lines.push(`${String(i * 10).padStart(3)}-${String(i * 10 + 9).padStart(3)}: ${bar} ${hist[i]}`);
  }

  if (anomalies.length) {
    lines.push(`\n## 异常清单`);
    for (const a of anomalies.slice(0, 100)) lines.push(`- ${a}`);
    if (anomalies.length > 100) lines.push(`- ...（其余 ${anomalies.length - 100} 条省略）`);
  }

  mkdirSync(dirname(reportPath), { recursive: true });
  writeFileSync(reportPath, lines.join('\n'), 'utf8');
  console.log(lines.join('\n'));

  if (unsolvable > 0 || dupCount > 0) {
    console.error('\n✗ 存在不可解或重复关卡');
    process.exit(1);
  }
  console.log('\n✓ 体检通过');
}

main();
```
