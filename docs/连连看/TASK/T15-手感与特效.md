### 15.1 写 `src/render/fx.ts`（粒子与动画系统）

```ts
/** 粒子。 */
interface Particle {
  x: number; y: number;
  vx: number; vy: number;
  life: number; maxLife: number;
  size: number;
  color: string;
}

/** 波次点亮动画的一层。 */
interface WaveLayer {
  cells: number[];
  startAt: number;
}

/**
 * 特效系统。管理粒子、通关波次、皮肤解锁光晕。
 */
export class Fx {
  private particles: Particle[] = [];
  private waves: WaveLayer[] = [];
  private bursts: Array<{ x: number; y: number; at: number; color: string }> = [];

  /**
   * 生成粒子迸发。
   */
  burst(x: number, y: number, color: string, count = 16): void {
    for (let i = 0; i < count; i++) {
      const angle = (Math.PI * 2 * i) / count + Math.random() * 0.4;
      const speed = 120 + Math.random() * 180;
      this.particles.push({
        x, y,
        vx: Math.cos(angle) * speed,
        vy: Math.sin(angle) * speed,
        life: 0,
        maxLife: 500 + Math.random() * 300,
        size: 3 + Math.random() * 4,
        color,
      });
    }
  }

  /** 通关波次：把路径按距离分层。 */
  wave(cells: number[]): void {
    const layers: number[][] = [];
    for (let i = 0; i < cells.length; i++) {
      const layerIdx = Math.floor(i / 3);
      if (!layers[layerIdx]) layers[layerIdx] = [];
      layers[layerIdx].push(cells[i]);
    }
    const now = Date.now();
    layers.forEach((cellsInLayer, i) => {
      this.waves.push({ cells: cellsInLayer, startAt: now + i * 60 });
    });
  }

  /** 记录一次位置迸发（渲染时再取位置）。 */
  burstAt(idx: number, w: number, cell: number, originX: number, originY: number, color: string): void {
    this.bursts.push({
      x: originX + (idx % w) * cell + cell / 2,
      y: originY + Math.floor(idx / w) * cell + cell / 2,
      at: Date.now(),
      color,
    });
  }

  /**
   * 更新并绘制所有特效。
   * @param dt 距上次更新的毫秒数
   */
  update(ctx: CanvasRenderingContext2D, dt: number): void {
    const now = Date.now();

    // 粒子
    for (let i = this.particles.length - 1; i >= 0; i--) {
      const p = this.particles[i];
      p.life += dt;
      if (p.life >= p.maxLife) { this.particles.splice(i, 1); continue; }
      p.x += (p.vx * dt) / 1000;
      p.y += (p.vy * dt) / 1000;
      p.vy += 400 * (dt / 1000);   // 重力
    }
    ctx.save();
    for (const p of this.particles) {
      const t = 1 - p.life / p.maxLife;
      ctx.globalAlpha = t;
      ctx.fillStyle = p.color;
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.size * t, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();

    // 位置迸发
    ctx.save();
    for (let i = this.bursts.length - 1; i >= 0; i--) {
      const b = this.bursts[i];
      const elapsed = now - b.at;
      if (elapsed > 600) { this.bursts.splice(i, 1); continue; }
      const r = (elapsed / 600) * 60;
      ctx.globalAlpha = 1 - elapsed / 600;
      ctx.strokeStyle = b.color;
      ctx.lineWidth = 3;
      ctx.beginPath();
      ctx.arc(b.x, b.y, r, 0, Math.PI * 2);
      ctx.stroke();
    }
    ctx.restore();
  }

  /** 当前是否还有活跃特效。 */
  get active(): boolean {
    return this.particles.length > 0 || this.waves.length > 0 || this.bursts.length > 0;
  }

  /** 清空。 */
  clear(): void {
    this.particles = [];
    this.waves = [];
    this.bursts = [];
  }
}
```

### 15.2 写 `src/render/audio.ts`（WebAudio 合成音效）

```ts
/**
 * 音效合成器。不使用音频文件，全部用 WebAudio 生成。
 * 微信端不可用 WebAudio → 由 platform.vibrate() 兜底。
 */
export class Audio {
  private ctx: AudioContext | null = null;

  /** 惰性创建 AudioContext（必须在用户手势后调用）。 */
  private ensure(): AudioContext | null {
    if (this.ctx) return this.ctx;
    try {
      const Ctor = (window.AudioContext ?? (window as any).webkitAudioContext);
      if (!Ctor) return null;
      this.ctx = new Ctor();
      return this.ctx;
    } catch {
      return null;
    }
  }

  /** 播放一个音。 */
  private tone(freq: number, durationMs: number, type: OscillatorType, volume: number, delayMs = 0): void {
    const ctx = this.ensure();
    if (!ctx) return;
    if (ctx.state === 'suspended') void ctx.resume();
    const t0 = ctx.currentTime + delayMs / 1000;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(volume, t0);
    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + durationMs / 1000);
    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start(t0);
    osc.stop(t0 + durationMs / 1000 + 0.02);
  }

  /** 点亮砖块。 */
  light(): void { this.tone(880, 30, 'square', 0.15); }

  /** 断线。 */
  break(): void {
    this.tone(220, 120, 'square', 0.18);
    this.tone(200, 60, 'square', 0.12, 40);
  }

  /** 通关琶音。 */
  win(): void {
    this.tone(523, 100, 'triangle', 0.2, 0);
    this.tone(659, 100, 'triangle', 0.2, 100);
    this.tone(784, 180, 'triangle', 0.22, 200);
  }

  /** 解锁皮肤。 */
  unlock(): void {
    this.tone(1046, 150, 'sine', 0.18, 0);
    this.tone(1568, 200, 'sine', 0.2, 120);
  }
}

/** 全局单例。 */
export const audio = new Audio();
```

```ts
/** 缓动函数集。 */
export const Ease = {
  /** 二次缓出。 */
  outQuad: (t: number) => 1 - (1 - t) * (1 - t),
  /** 三次缓出。 */
  outCubic: (t: number) => 1 - Math.pow(1 - t, 3),
  /** 二次缓入。 */
  inQuad: (t: number) => t * t,
  /** 弹性缓出。 */
  outBack: (t: number) => {
    const c1 = 1.70158, c3 = c1 + 1;
    return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2);
  },
  /** 阻尼正弦（用于抖动）。 */
  damped: (t: number, freq = 6) => Math.sin(t * Math.PI * freq) * (1 - t),
};
```

### 15.4 接入 `src/main.ts`：动画与反馈

```ts
import { Fx } from './render/fx';
import { audio } from './render/audio';

const fx = new Fx();
let lastFrameAt = Date.now();
let winAnimStart = 0;

/** 记录上次帧时间。 */
function tick() {
  const now = Date.now();
  const dt = Math.min(50, now - lastFrameAt);
  lastFrameAt = now;
  return { now, dt };
}
```

**通关波次的格子高亮**：在 `drawBoard` 的 state 里增加 `waveLit: Set<number>`，由 main 循环根据 `fx` 的波次时间更新。简化实现：在 `main.ts` 里维护

```ts
/** 波次点亮状态。 */
let waveLit = new Set<number>();

function updateWave(now: number) {
  waveLit = new Set();
  // 距离终点越近的点亮越早
  const st = session.state;
  if (!st || !winAnimStart) return;
  const elapsed = now - winAnimStart;
  const total = st.path.length;
  for (let i = 0; i < total; i++) {
    // 从终点往回扩散
    const delay = (total - 1 - i) * 60;
    if (elapsed >= delay) waveLit.add(st.path[i]);
  }
}
```
把 `waveLit` 合并进 `lit` 传给 `drawBoard`。

```markdown
# 手感与反馈规范

## 1. 反馈时序表

| 事件 | 视觉反馈 | 时长 | 音效 | 触觉 |
|---|---|---|---|---|
| 按下起点 | 起点放大 1.1× | 80ms | 无 | 无 |
| 点亮砖块 | 砖块变亮 + 顶部高光 + 4 颗粒子迸发 | 120ms | 方波 880Hz 30ms | 无 |
| 断线 | 线头横向抖动 ±4px 3 次 + 红色闪烁 | 200ms | 方波 220Hz 120ms | 短振动 |
| 进入单向块（合法） | 箭头轻微脉冲 | 150ms | 无 | 无 |
| 尝试进入非法格 | 无变化（静默拒绝） | — | 无 | 无 |
| 提示出现 | 目标格黄色描边 + 呼吸动画（1.2s 循环） | 持续 | 无 | 无 |
| 松手未到终点 | 全部砖块熄灭 | 150ms | 无 | 无 |
| 抵达终点 | 冻结输入 + 从终点反向扩散点亮波次（每层 60ms） | ≤1.2s | C-E-G 琶音 | 无 |
| 结算页入场 | 底部滑入 + 背景遮罩淡入 | 300ms | 无 | 无 |
| 皮肤解锁 | 卡片弹入 + 光晕扩散 + 皮肤预览旋转 | 400ms | 1046→1568Hz | 短振动 |
| 进度条填充 | 从左到右填充 | 500ms | 无 | 无 |

## 2. 原则

1. **每个动作 0.1 秒内必须有视觉反馈** —— 拖动时每点亮一格都要有响应
2. **非法操作静默拒绝** —— 不弹窗、不变色、不打断，只在「断线」时给一次明确反馈
3. **失败即反馈** —— 断线要明确告知玩家"刚才那步不行"，但不能惩罚
4. **通关要爽** —— 波次点亮 + 音效 + 震动，三者同时

## 3. 动画参数

| 动画 | 时长 | 缓动 |
|---|---|---|
| 砖块点亮 | 120ms | easeOutQuad |
| 断线抖动 | 200ms | 阻尼正弦 (freq=6) |
| 通关波次 | 每层 60ms，总 ≤1.2s | easeOutCubic |
| 结算页入场 | 300ms | easeOutCubic |
| 皮肤解锁弹窗 | 400ms | easeOutBack |
| 进度条填充 | 500ms | easeOutCubic |

## 4. 触觉反馈（微信）

- 断线：`wx.vibrateShort({type:'medium'})`
- 通关：`wx.vibrateShort({type:'heavy'})`
- 皮肤解锁：`wx.vibrateShort({type:'light'})`
```

## 坑
- **WebAudio 必须在用户手势后创建/恢复** → `ensure()` 里检查 `ctx.state === 'suspended'` 并 `resume()`
- **微信端无 WebAudio** → `audio.ensure()` 里 `window` 未定义会抛错 → 已用 try/catch 包住
- **粒子用 `Math.random()`** → 特效层不要求确定性，可以直接用（`core/` 层才是硬性要求）
- **通关波次与结算页时序** → 结算页延后 1.2 秒，否则动画会被遮罩盖住
- `winAnimStart` 归零 → 下一关加载时重置
