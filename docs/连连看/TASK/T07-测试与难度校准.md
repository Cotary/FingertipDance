```ts
import { describe, it, expect } from 'vitest';
import { readFileSync, existsSync } from 'node:fs';
import { resolve } from 'node:path';
import { deserializeLevel, type LevelTableFile } from '../src/core/levelTable';
import { solve, isPathLegal } from '../src/core/solver';
import { targetDiff } from '../src/core/curve';

const LEVELS = resolve(__dirname, '../data/levels.json');
const SOLUTIONS = resolve(__dirname, '../data/solutions.json');

const hasFiles = existsSync(LEVELS) && existsSync(SOLUTIONS);
const describeIf = hasFiles ? describe : describe.skip;

describeIf('关卡表管线', () => {
  const table = JSON.parse(readFileSync(LEVELS, 'utf8')) as LevelTableFile;

  it('关卡数与 count 一致', () => {
    expect(table.count).toBe(table.levels.length);
    expect(table.count).toBe(1000);
  });

  it('关号连续且从 1 开始', () => {
    table.levels.forEach((l, i) => expect(l.idx).toBe(i + 1));
  });

  it('全部可解且 shortest 与求解器一致（抽检 100 关）', () => {
    const step = Math.floor(table.levels.length / 100);
    for (let i = 0; i < table.levels.length; i += step) {
      const s = table.levels[i];
      const level = deserializeLevel(s);
      const res = solve(level.grid);
      expect(res.solvable, `第 ${s.idx} 关应可解`).toBe(true);
      expect(res.steps, `第 ${s.idx} 关步数应一致`).toBe(s.shortest);
    }
  });

  it('全部解法合法（抽检 100 关）', () => {
    const sol = JSON.parse(readFileSync(SOLUTIONS, 'utf8'));
    const map = new Map(sol.levels.map((s: any) => [s.idx, s]));
    const step = Math.floor(table.levels.length / 100);
    for (let i = 0; i < table.levels.length; i += step) {
      const s = table.levels[i];
      const level = deserializeLevel(s);
      const sv = map.get(s.idx);
      expect(sv).toBeDefined();
      expect(isPathLegal(level.grid, sv.solution)).toBe(true);
      expect(sv.solution.length - 1).toBe(s.shortest);
    }
  });

  it('无重复布局', () => {
    const fps = new Set(table.levels.map((l) => `${l.w}x${l.h}:${l.grid}`));
    expect(fps.size).toBe(table.levels.length);
  });

  it('难度与目标偏差总体可控（平均 ≤ 8）', () => {
    let sum = 0;
    for (const l of table.levels) sum += Math.abs(l.diff - targetDiff(l.idx));
    expect(sum / table.levels.length).toBeLessThanOrEqual(8);
  });

  it('后期关卡唯一解占比 ≥ 60%', () => {
    const late = table.levels.filter((l) => l.idx > 300);
    const sol = JSON.parse(readFileSync(SOLUTIONS, 'utf8'));
    const map = new Map(sol.levels.map((s: any) => [s.idx, s]));
    // 用求解器统计
    let unique = 0;
    for (const l of late) {
      const level = deserializeLevel(l);
      if (solve(level.grid).pathCount <= 1) unique++;
    }
    expect(unique / late.length).toBeGreaterThanOrEqual(0.6);
  });
});
```

### 7.2 写 `test/edge.test.ts`（边界与健壮性）

```ts
import { describe, it, expect } from 'vitest';
import { solve, isPathLegal, findHint, solveFrom } from '../src/core/solver';
import { generateLevel } from '../src/core/generator';
import { encodeGrid, decodeGrid } from '../src/core/codec';
import { CellType, Dir, type Cell } from '../src/core/types';

describe('边界情况', () => {
  it('空网格不崩溃', () => {
    expect(() => solve([])).not.toThrow();
    expect(solve([]).solvable).toBe(false);
  });

  it('没有终点时返回不可解', () => {
    const g: Cell[][] = [[{ type: CellType.START, dir: Dir.UP }]];
    expect(solve(g).solvable).toBe(false);
  });

  it('起点等于终点时步数为 0', () => {
    const g: Cell[][] = [[{ type: CellType.START, dir: Dir.UP }]];
    const r = solve(g);
    expect(r.solvable).toBe(true);
    expect(r.steps).toBe(0);
  });

  it('isPathLegal 拒绝空路径与单点路径', () => {
    const g: Cell[][] = [[{ type: CellType.START, dir: Dir.UP }, { type: CellType.END, dir: Dir.UP }]];
    expect(isPathLegal(g, [])).toBe(false);
    expect(isPathLegal(g, [0])).toBe(false);
  });

  it('isPathLegal 拒绝起点终点出现在中间', () => {
    const g: Cell[][] = [
      [{ type: CellType.START, dir: Dir.UP }, { type: CellType.BRICK, dir: Dir.UP }],
      [{ type: CellType.BRICK, dir: Dir.UP }, { type: CellType.END, dir: Dir.UP }],
    ];
    // 路径经过 START 两次
    expect(isPathLegal(g, [0, 1, 0, 3])).toBe(false);
  });

  it('编码遇到尺寸不符会抛错', () => {
    expect(() => encodeGrid([], 3, 3)).toThrow();
    expect(() => decodeGrid('AAAA', 2, 2)).toThrow();
  });

  it('findHint 在走岔时返回空数组而非崩溃', () => {
    const g: Cell[][] = [
      [{ type: CellType.START, dir: Dir.UP }, { type: CellType.BLOCK, dir: Dir.UP }],
      [{ type: CellType.BLOCK, dir: Dir.UP }, { type: CellType.END, dir: Dir.UP }],
    ];
    expect(() => findHint(g, { x: 0, y: 0 })).not.toThrow();
  });

  it('solveFrom 对不可达点返回 null', () => {
    const g: Cell[][] = [
      [{ type: CellType.START, dir: Dir.UP }, { type: CellType.BLOCK, dir: Dir.UP }],
      [{ type: CellType.BLOCK, dir: Dir.UP }, { type: CellType.END, dir: Dir.UP }],
    ];
    expect(solveFrom(g, 1, 1)).toBeNull();
  });

  it('生成器在极端关号仍能产出可解关卡', () => {
    for (const idx of [1, 500, 1000]) {
      const lv = generateLevel(idx, 999);
      expect(solve(lv.grid).solvable).toBe(true);
    }
  });
});
```

### 7.4 难度校准循环（**人工操作，必须做**）

```
循环：
  1. npm run level:gen -- --count 1000 --seed 20261101 --out data/levels.json
  2. npm run level:audit -- --file data/levels.json --report out/audit.md
  3. 打开 out/audit.md，看「每 10 关一档」表
  4. 若「平均diff」与「目标diff」偏差 > 8：
       a. 偏差为正（实际难度高于目标）→ 提高 curve.ts 的 openRatio 或降低 blocks
       b. 偏差为负（实际难度低于目标）→ 降低 openRatio 或提高 blocks，或提高 uniqueness 权重
     改完回到步骤 1
  5. 若「唯一解占比」< 60%：提高 curve.ts 的 branchPenalty 上限，或加强 generator 的 narrowBranches
  6. 若「平均步数」增长过快（相邻档增量 > 5）：说明尺寸/阻挡变化太激进，调缓分段表
  7. 最多迭代 5 轮，达标即停
```

**达标标准（全部必须满足）**
- 不可解 = 0
- 重复布局 = 0
- 平均 diff 偏差 ≤ 8
- 后期唯一解占比 ≥ 60%
- 相邻档平均 diff 增量 ∈ [3, 12]
- 相邻档平均步数增量 ∈ [1, 5]
